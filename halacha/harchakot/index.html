הקדמה

1/26

בִּזְמַן שֶׁהָאִשָּׁה בְּטֻמְאַת נִדָּתָהּ נֶאֱסַר עַל בְּנֵי הַזּוּג לְהִתְקָרֵב זֶה לָזוֹ וְזוֹ לָזֶה, וְאִסּוּרִים אֵלּוּ נֶחְלָקִים לִשְׁנֵי סוּגִים כְּלָלִיִּים:

א. אִסּוּר תַּשְׁמִישׁ, שֶׁהוּא אִסּוּר תּוֹרָה, שֶׁנֶּאֱמַר "וְאֶל אִשָּׁה בְּנִדַּת טֻמְאָתָהּ לֹא תִקְרַב לְגַלּוֹת עֶרְוָתָהּ" (ויקרא יח, יט), וְעָנְשׁוֹ כָּרֵת לָעוֹבְרִים בְּמֵזִיד.

עֹנֶשׁ הַכָּרֵת כּוֹלֵל ג' דְּבָרִים חֲמוּרִים מְאֹד:

  1. א) מָוֶת בְּקִצּוּר שָׁנִים [עַד גִּיל 50];
  2. ב) מָוֶת לְלֹא יְלָדִים [שֶׁלֹּא יִוָּלְדוּ לוֹ יְלָדִים וְאִם כְּבָר נוֹלְדוּ – יָמוּתוּ בְּחַיָּיו, ר"ל];
  3. ג) הַנֶּפֶשׁ נִכְרֶתֶת מֵחַיֵּי הָעוֹלָם הַבָּא [מִגַּן עֵדֶן].

הַבֵּית יוֹסֵף (סי' קצה) מֵבִיא, כִּי אִסּוּר נִדָּה נִכְלָל בְּאִסּוּר גִּלּוּי עֲרָיוֹת שֶׁעֲלֵיהֶם נֶאֱמַר "יֵהָרֵג וְאַל יַעֲבֹר", כְּלוֹמַר, אִם כּוֹפִים גֶּבֶר לָבוֹא עַל [אִשְׁתּוֹ] נִדָּה וְאִם לֹא יַעֲשֶׂה כֵּן יַהַרְגוּהוּ – עָלָיו לִמְסֹר נַפְשׁוֹ לְמִיתָה וְלֹא לַעֲבֹר [אִשָּׁה אֵינָהּ צְרִיכָה לִמְסֹר עַצְמָהּ לְמִיתָה בָּזֶה, כִּי הִיא אֵינָהּ עוֹשָׂה שׁוּם פְּעֻלָּה מִצִּדָּהּ].

וְכָתַב אַדְמוּ"ר הַזָּקֵן (תניא, אגרת התשובה פ"ו), שֶׁבִּזְמַן הַגָּלוּת אֶפְשָׁר שֶׁהַחוֹטֵא יַאֲרִיךְ יָמִים, כִּי אע"פ שֶׁחַיָּיו נִכְרָתִים מִשָּׁרְשָׁם בַּקְּדֻשָּׁה, מ"מ יָכוֹל לְקַבֵּל חַיּוּת מִצַּד הַקְּלִפָּה. אוּלָם וַדַּאי שֶׁבְּיַד הָאָדָם לָשׁוּב בִּתְשׁוּבָה שְׁלֵמָה מֵעֹמֶק הַלֵּב, לְסַלֵּק הַשְׁפָּעַת הַקְּלִפָּה מֵעָלָיו וְלַחֲזֹר וּלְקַבֵּל חַיּוּת מֵהַקְּדֻשָּׁה, וּלְבַטֵּל מֵעָלָיו אֶת הָעֳנָשִׁים הַנַּ"ל.

וְדַע, שֶׁבְּנוֹסָף לְעֹנֶשׁ הָאִישׁ וְהָאִשָּׁה, חֵטְא זֶה גַּם פּוֹגֵם בַּיְלָדִים שֶׁנּוֹלְדוּ שֶׁלֹּא בְּטָהֳרָה וְגוֹרֵם לָהֶם טֶבַע רַע וּמְגֻשָּׁם [וְגַם עָלוּל לִגְרֹם לְחִסָּרוֹן בִּבְרִיאוּת הַגּוּף וְהַנֶּפֶשׁ, ח"ו (מסכת כלה רבתי פ"ב מ"ט. ראה רמב"ם הל' איסורי ביאה פט"ו ה"א)].

הָרַבִּי (ראה ליקוטי שיחות ח"ב עמ' 579. וראה גם באגרות קודש חי"ח עמ' תנב) גַּם מוֹצֵא בְּכָךְ אֶת אֶחָד הַהֶסְבֵּרִים לָעֻבְדָּה, שֶׁבַּדּוֹרוֹת הָאַחֲרוֹנִים יֶשְׁנָם בְּעַם יִשְׂרָאֵל אֲנָשִׁים שֶׁחוֹטְאִים בִּדְבָרִים שֶׁבֶּעָבָר יְהוּדִים לֹא הָיוּ נִכְשָׁלִים בָּהֶם, שֶׁזֶּה נִגְרָם מִפְּנֵי שֶׁהוֹרֵיהֶם לֹא שָׁמְרוּ עַל טָהֳרַת הַמִּשְׁפָּחָה וְכָךְ נוֹלְדוּ לָהֶם יְלָדִים גַּסִּים וּמְגֻשָּׁמִים, וְהַדְּבָרִים מַרְאִים עַד כַּמָּה חָשׁוּב שֶׁיְּצִירַת הָעֻבָּר תִּהְיֶה בְּטָהֳרָה.

אָמְנָם נִתָּן לַעֲשׂוֹת "תִּקּוּן" מְסֻיָּם גַּם כְּלַפֵּי הַיֶּלֶד, בְּכָךְ שֶׁהַהוֹרִים יַקְפִּידוּ מִכָּאן וְאֵילָךְ עַל הִלְכוֹת טָהֳרַת הַמִּשְׁפָּחָה וְאַף יָפִיצוּ אֶת הַמִּצְוָה הַזֹּאת.

בְּנוֹסָף לְעִנְיַן הַטָּהֳרָה חָשׁוּב מְאֹד גַּם עִנְיַן הַקְּדֻשָּׁה דְּהַיְנוּ שֶׁיְּצִירַת הָעֻבָּר תִּהְיֶה מִתּוֹךְ מַחֲשָׁבוֹת טוֹבוֹת וּקְדוֹשׁוֹת שֶׁל הוֹרָיו, כַּמּוּבָא בְּסֵפֶר הַתַּנְיָא (לקוטי אמרים פ"ב) בְּשֵׁם הַזֹּהַר (ח"ב קיב, א. ח"ג מט, ב. ובכ"מ), שֶׁהָעִקָּר תָּלוּי בָּזֶה שֶׁיְּקַדְּשׁוּ הַהוֹרִים עַצְמָם בִּזְמַן הַתַּשְׁמִישׁ שֶׁעי"ז מַמְשִׁיכִים לְבוּשׁ קָדוֹשׁ [כְּלוֹמַר, תְּכוּנוֹת שֶׁל קְדֻשָּׁה] לְנִשְׁמוֹת הַבָּנִים כו'.

ב. אִסּוּר נוֹסָף מִן הַתּוֹרָה הוּא קֵרוּב לָאִשָּׁה [בְּדֶרֶךְ חִבָּה], כְּמוֹ חִבּוּק וְנִשּׁוּק. גַּם אִסּוּר זֶה נִלְמָד מִדִּיּוּק לְשׁוֹן הַפָּסוּק "וְאֶל אִשָּׁה בְּנִדַּת טֻמְאָתָהּ לֹא תִקְרַב לְגַלּוֹת עֶרְוָתָהּ" (ויקרא יח, יט), כְּלוֹמַר, גַּם הַקִּרְבָה הַגּוּפָנִית נֶאֶסְרָה מֵאַחַר שֶׁעֲלוּלָה לְהָבִיא לִידֵי גִּלּוּי עֶרְוָה מַמָּשׁ [תַּשְׁמִישׁ].

חֲזַ"ל הוֹסִיפוּ הַרְחָקוֹת רַבּוֹת כְּדֵי שֶׁלֹּא יָבוֹאוּ לִידֵי קֵרוּב וְתַשְׁמִישׁ וְגַם הֵם חֲמוּרִים מְאֹד וּמִי שֶׁאֵינוֹ נִזְהָר בָּזֶה עָלוּל לְהִכָּשֵׁל בְּתַשְׁמִישׁ, שֶׁהוּא אִסּוּר כָּרֵת, כַּנַּ"ל.

סיפור מהגמרא

2/26

בַּתַּלְמוּד (שבת יג, א) מוּבָא מַעֲשֶׂה בְּתַלְמִיד־חָכָם אֶחָד שֶׁנִּפְטַר בְּגִיל צָעִיר, ל"ע, וְהָיְתָה אִשְׁתּוֹ נוֹטֶלֶת אֶת הַתְּפִלִּין שֶׁלּוֹ וְסוֹבֶבֶת בְּבָתֵּי־כְּנֵסִיּוֹת וּבְבָתֵּי־מִדְרָשׁוֹת וְטוֹעֶנֶת, שֶׁהֲרֵי נֶאֱמַר עַל הַתּוֹרָה "כִּי הוּא חַיֶּיךָ וְאֹרֶךְ יָמֶיךָ" (דברים ל, כ), בַּעְלָהּ שֶׁלָּמַד הַרְבֵּה תּוֹרָה, הֵיכָן אוֹתָהּ אֲרִיכוּת יָמִים הַמֻּבְטַחַת לוֹ? וְלֹא הָיָה אָדָם שֶׁהֱשִׁיבָהּ תְּשׁוּבָה עַל טַעֲנָה זוֹ, עַד שֶׁהִתְגַּלָּה אֵלֶיהָ אֵלִיָּהוּ הַנָּבִיא וְאַחַר שֶׁהֵשִׂיחָה בְּפָנָיו כָּל הַמְּאֹרָע, שָׁאַל אוֹתָהּ: "בִּתִּי, כֵּיצַד נָהַג עִמָּךְ בִּזְמַן הַנִּדָּה?" עָנְתָה הָאִשָּׁה, שֶׁנָּהַג עִמָּהּ כָּל הַהַרְחָקוֹת הַכְּתוּבוֹת בַּהֲלָכָה, כְּגוֹן שֶׁלֹּא אָכַל עִמָּהּ בְּאוֹתוֹ שֻׁלְחָן [אִם לֹא הָיָה דָּבָר שֶׁל הֶכֵּר בֵּינֵיהֶם] וְכוּ'.

שׁוּב שְׁאָלָהּ אֵלִיָּהוּ הַנָּבִיא: "בִּימֵי לִבּוּנֵךְ כֵּיצַד נָהַג?" כְּלוֹמַר, כֵּיצַד הִתְנַהֵג עִמָּךְ בִּזְמַן שִׁבְעָה נְקִיִּים [שֶׁאָז הָאִשָּׁה לוֹבֶשֶׁת לְבָנִים]?

עָנְתָה, שֶׁבְּאוֹתָם יָמִים לֹא הִקְפִּיד עַל הַהַרְחָקוֹת בַּסְּעוּדָה וְכוּ' אֲבָל וַדַּאי נִזְהַר מִתַּשְׁמִישׁ כו'.

עָנָה לָהּ אֵלִיָּהוּ הַנָּבִיא: "בָּרוּךְ הַמָּקוֹם שֶׁהֲרָגוֹ, שֶׁהֲרֵי נֶאֱמַר 'וְאֶל אִשָּׁה בְּנִדַּת טֻמְאָתָהּ לֹא תִקְרַב' גו'". כְּלוֹמַר, אֵלִיָּהוּ הַנָּבִיא הִצְדִּיק אֶת הָעֹנֶשׁ הֶחָמוּר שֶׁלּוֹ בְּזֶה שֶׁטָּעָה בַּהֲלָכָה וְלֹא נִזְהַר בַּהַרְחָקוֹת בְּשִׁבְעָה נְקִיִּים.

מִזֶּה רוֹאִים עַד הֵיכָן הַדְּבָרִים מַגִּיעִים! אֵלִיָּהוּ אַף הִשְׁתַּמֵּשׁ בַּבִּטּוּי "בָּרוּךְ הַמָּקוֹם שֶׁהֲרָגוֹ!", כְּלוֹמַר, טוֹבָה נַעֲשֵׂית לוֹ שֶׁנֶּעֱנַשׁ בעוה"ז בְּמִיתָה בְּגִיל צָעִיר כְּדֵי שֶׁיִּזְכֶּה לָעוֹלָם הַבָּא.

האיסור הכללי

א. אִשָּׁה שֶׁנֶּאֶסְרָה עַל בַּעְלָהּ בְּאֶחָד מֵהָאֳפָנִים הַמְבֹאָרִים לְעֵיל בְּפֶרֶק א' – הֲרֵי הִיא אֲסוּרָה בְּבִיאָה וּבִנְגִיעָה [וּמֵחֲמַת זֶה בְּעוֹד דְּבָרִים שֶׁיִּתְבָּאֲרוּ לְהַלָּן]. אַךְ מוּבָן שֶׁיֵּשׁ לְבַעְלָהּ לְהִתְיַחֵס אֵלֶיהָ בְּכָבוֹד וְאַף בְּיֶתֶר רְגִישׁוּת בְּאוֹתוֹ הַזְּמַן בָּאֳפָנִים הַמֻּתָּרִים.

אִם יֵשׁ מָנָה הַמֻּרְכֶּבֶת מִכַּמָּה פְּרִיטִים, כְּגוֹן בָּשָׂר וְתוֹסָפוֹת וְהָאִשָּׁה אָכְלָה אֶת הַבָּשָׂר וְלֹא טָעֲמָה מֵהַתּוֹסָפוֹת, אָסוּר לַבַּעַל לֶאֱכֹל מֵהַתּוֹסָפוֹת, שֶׁכֵּן הַכֹּל נֶחְשָׁב כִּשְׁיָרֶיהָ13.

יד. אִם הִשְׁאִירָה שְׁיָרַיִם בַּכּוֹס אוֹ בַּצַּלַּחַת, וְשָׁתָה אָדָם נוֹסָף מֵהַכּוֹס אוֹ אָכַל אָדָם נוֹסָף מֵהַצַּלַּחַת אַחֲרֵי אֲכִילַת הָאִשָּׁה אוֹ שֶׁהֶעֱבִירוּ לְכוֹס אוֹ צַלַּחַת אַחֶרֶת, גַּם אִם הֶחֱזִירוּהוּ לַכְּלִי הָרִאשׁוֹן, אוֹ שֶׁיָּצְאָה הָאִשָּׁה מֵהַחֶדֶר – מֻתָּר הַבַּעַל לֶאֱכֹל אֶת הַשְּׁיָרַיִם. וְכֵן אִם אֵינוֹ יוֹדֵעַ שֶׁהִיא שָׁתְתָה אוֹ אָכְלָה מִזֶּה – אֵינָהּ צְרִיכָה לְהַגִּיד לוֹ. אֲבָל אִם יוֹדֵעַ שֶׁשָּׁתְתָה וְרַק שֶׁאֵינוֹ יוֹדֵעַ שֶׁנִּטְמְאָה – חַיֶּבֶת לְהוֹדִיעַ לוֹ. וְכֵן אִם שָׁתְתָה כָּל תְּכוּלַת הַכּוֹס וְחָזְרוּ וּמִלְּאוּהוּ – מֻתָּר אע"פ שֶׁלֹּא הֱדִיחוּהוּ, אֲבָל מִדַּת חֲסִידוּת לְהָדִיחַ הַכּוֹס לִפְנֵי הַשִּׁמּוּשׁ14.

כ. אָסוּר לַבַּעַל לֵישֵׁב וְלִשְׁכַּב עַל מִטַּת אִשְׁתּוֹ גַּם שֶׁלֹּא בְּפָנֶיהָ. כֵּן אָסוּר לְהִשְׁתַּמֵּשׁ בַּמַּצָּעִים הַמְיֻחָדִים לָהּ [כְּגוֹן אִם יֵשׁ כַּר אוֹ שְׂמִיכָה הַמְיֻחָדִים לָהּ].

אֲבָל לָאִשָּׁה מֻתָּר לֵישֵׁב עַל מִטָּתוֹ אֲפִלּוּ בְּפָנָיו וְכֵן לֵישֵׁב עַל מַצָּעָיו אֲפִלּוּ בְּפָנָיו, אֲבָל אֲסוּרָה לִשְׁכַּב עַל מִטָּתוֹ בְּפָנָיו. וְאִם הָאִשָּׁה בְּמָקוֹם מְרֻחָק [כְּשָׁעָה נְסִיעָה וָיוֹתֵר] – מֻתָּר לֵישֵׁב עַל מִטָּתָהּ. וְאִם הִיא מֵעֵבֶר לַיָּם אוֹ בְּאֶרֶץ רְחוֹקָה, אוֹ שֶׁנָּסְעָה לְכַמָּה יָמִים מֵהַבַּיִת – מֻתָּר גַּם לִשְׁכַּב עַל מִטָּתָהּ.

"מִטָּה שֶׁלָּהּ" נִקְרֵאת אִם הִיא יְשֵׁנָה עָלֶיהָ בִּקְבִיעוּת אוֹ הִתְחִילָה לִישֹׁן עָלֶיהָ וּבְכַוָּנָתָהּ שֶׁתִּישַׁן בָּהּ בִּקְבִיעוּת. אֲבָל אִם יָשְׁנָה עָלֶיהָ לַיְלָה אֶחָד בְּאַקְרַאי – אֵין אִסּוּר [וְיֵשׁ שֶׁהִתִּירוּ לַבַּעַל לֵישֵׁב עַל מִטַּת אִשְׁתּוֹ הַיּוֹלֶדֶת בְּבֵית־הָרְפוּאָה, אִם הָאִשָּׁה אֵינָהּ בַּמִּטָּה, מִכֵּיוָן שֶׁאי"ז לִזְמַן אָרֹךְ וְאֵינָהּ מְיֹעֶדֶת לְחַיֵּי מִשְׁפָּחָה וְלֹא נִקְרֵאת "מִטָּתָהּ". וְיֵשׁ שֶׁאָסְרוּ גַּם בָּזֶה]20.

כא. מִטָּה שֶׁבַּיּוֹם מִתְקַפֶּלֶת וְהוֹפֶכֶת לְכִסֵּא נוֹחַ וכיו"ב – מֻתָּר לַבַּעַל לֵישֵׁב עָלֶיהָ אַחַר שֶׁקִּפְּלוּהָ לְכִסֵּא.

וְאִם יֵשׁ לָהֶם מִטּוֹת שֶׁמַּכְנִיסִים בַּיּוֹם אֶת שְׁנֵי הַמִּזְרָנִים אֶחָד עַל הַשֵּׁנִי, הֲרֵי אִם הַמִּזְרָן שֶׁל הַבַּעַל לְמַעְלָה – מֻתָּר לֵישֵׁב עָלָיו, וְאִם הַמִּזְרָן שֶׁל הָאִשָּׁה לְמַעְלָה – נֶחְלְקוּ הַפּוֹסְקִים בָּזֶה.

וּמ"מ מֻתָּר לַבַּעַל לִנְגֹּעַ בְּמִטָּתָהּ אוֹ בִּכְלֵי מִטָּתָהּ אוֹ בִּבְגָדֶיהָ, אֲבָל אִם זֶה מֵבִיא לִידֵי הִרְהוּר יִמָּנַע מִכָּךְ [וּלְעִנְיַן הַצָּעַת מִטָּה בִּזְמַן הָאִסּוּר – רְאֵה לְקַמָּן הֲלָכָה כט]21.

  2. בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סל"ב וְאֵילָךְ וּבְסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פכ"ח סכ"ב הוּבְאָה מַחְלֹקֶת הַאִם מֻתָּר לֵישֵׁב ע"ג סַפָּה כְּפוּלָה שֶׁבַּלַּיְלָה נִפְתַּחַת וּבַיּוֹם נִסְגֶּרֶת וּמַנִּיחִים אֶת מִזְרַן הָאִשָּׁה לְמַעְלָה, אֲבָל אִם מִזְרַן הָאִישׁ הוּא לְמַעְלָה – וַדַּאי מֻתָּר, כְּמ"ש שָׁם בַּהֶעָרָה.

כב. אֲסוּרִים לֵישֵׁב עַל סַפְסָל אוֹ מִטָּה מִתְנַדְנֶדֶת, כְּלוֹמַר שֶׁאֶחָד מַרְגִּישׁ בְּכֹבֶד הַשֵּׁנִי, שֶׁמִּפְּנֵי כָּבְדּוֹ הַמִּטָּה שׁוֹקַעַת קְצָת – בֵּין אִם זֶה מִזְרָן אֶחָד אוֹ שֶׁמְּחֻלָּק לִשְׁתֵּי מִזְרָנִים, אִם הָאֶחָד מַרְגִּישׁ שֶׁזֶּה שׁוֹקֵעַ בִּגְלַל כֹּבֶד הַשֵּׁנִי – אָסוּר. אֲבָל אִם בְּסִיס הַמִּטָּה יַצִּיב וְאֵינוֹ שׁוֹקֵעַ – מֻתָּר. וְגַם בְּמִטָּה שׁוֹקַעַת אִם הֵם יוֹשְׁבִים בְּרִחוּק גָּדוֹל כָּךְ שֶׁאֵין הָאֶחָד מַרְגִּישׁ בְּכָבְדּוֹ שֶׁל הַשֵּׁנִי – מֻתָּר.

וּמַתִּירִים (בִּשְׁעַת הַדְּחָק) כְּשֶׁאָדָם יוֹשֵׁב בֵּינֵיהֶם [בִּשְׁעַת הַדְּחָק גָּדוֹל מַתִּירִים גַּם אִם שָׂמִים בֵּינֵיהֶם חֵפֶץ גָּדוֹל, כְּגוֹן תִּיק גָּדוֹל]22.

כג. לְפִיכָךְ, בִּנְסִיעָה בִּמְכוֹנִית אוֹ בְּאוֹטוֹבּוּס וכיו"ב, אִם זֶה מוֹשָׁב אֶחָד – יֵשׁ לִפְעָמִים אִסּוּר מִצַּד הַדִּין [אִם הַמּוֹשָׁב שׁוֹקֵעַ כַּנַּ"ל]. וְאִם הַמּוֹשָׁבִים חֲלוּקִים וְאֵין שׁוֹקְעִים – אֵין אִסּוּר מִצַּד הַדִּין, אִם יְכוֹלִים לְהִזָּהֵר שֶׁלֹּא יִדָּחֲקוּ בְּיַחַד וְגַם לְהִזָּהֵר שֶׁלֹּא יִגְּעוּ הַמַּלְבּוּשִׁים זב"ז.

ע"כ, לְכַתְּחִלָּה יֵשְׁבוּ בִּשְׁנֵי מוֹשָׁבִים [כְּדֵי שֶׁלֹּא יָבוֹאוּ לִנְגִיעָה אוֹ נְגִיעַת הַבְּגָדִים]. אָמְנָם אִם ע"י שֶׁיֵּשְׁבוּ בְּנִפְרָד יֵצֵא שֶׁהַבַּעַל יֵשֵׁב לְיַד אִשָּׁה אַחֶרֶת אוֹ שֶׁהָאִשָּׁה תֵּשֵׁב לְיַד גֶּבֶר אַחֵר – עָדִיף שֶׁיֵּשְׁבוּ בְּנֵי הַזּוּג בְּאוֹתוֹ סַפְסָל [וְיָשִׂימוּ בֵּינֵיהֶם חֵפֶץ כְּדֵי שֶׁלֹּא יָבוֹאוּ לִידֵי נְגִיעָה].

וְכֵן בַּמָּטוֹס, שֶׁהַמּוֹשָׁבִים בּוֹ נִפְרָדִים, אִם אֶפְשָׁר יַשְׁאִירוּ מוֹשָׁב רֵיק בֵּינֵיהֶם. אֲבָל אִם הַמָּטוֹס מָלֵא וְאִם יֵשְׁבוּ בְּנִפְרָד יֵצֵא שֶׁהַבַּעַל יֵשֵׁב לְיַד אִשָּׁה אוֹ שֶׁהָאִשָּׁה תֵּשֵׁב לְיַד אִישׁ, אֲזַי יֵשְׁבוּ סְמוּכִים וְיִזָּהֲרוּ שֶׁלֹּא יִגְּעוּ זֶה בָּזֶה וְשֶׁבִּגְדֵיהֶם לֹא יִגְּעוּ בֵּינֵיהֶם23.

כד. לֹא יִסְעוּ יַחַד לְטִיּוּל בִּזְמַן הָאִסּוּר, אֲבָל מֻתָּרִים לִנְסֹעַ לְצֹרֶךְ, כְּגוֹן לְשִׂמְחָה, לְבִקּוּר קְרוֹבִים וּמַכָּרִים וכיו"ב. וְכֵן אִם מַטְּרַת הַנְּסִיעָה הִיא לְהוֹצִיא אֶת יַלְדֵיהֶם לְטִיּוּל – מֻתָּר [אִם הַמַּטָּרָה אֵינָהּ שֶׁהֵם עַצְמָם יְטַיְּלוּ].

וְאִם הָאִשָּׁה נוֹסַעַת לְצֹרֶךְ וְהַבַּעַל מִתְלַוֶּה כְּדֵי לְטַיֵּל – אָסוּר. וְאִם הָאִישׁ נוֹסֵעַ לְצֹרֶךְ וְהָאִשָּׁה מִתְלַוָּה אֵלָיו לְטִיּוּל – יֵשׁ לִשְׁאֹל רַב.

לא. אִם הַבַּעַל הוּא חוֹלֶה וְתָשׁוּשׁ, כָּךְ שֶׁאֵין חֲשָׁשׁ שֶׁיָּבוֹא לְתשהמ"ט בְּאִסּוּר, וְאֵין מִי שֶׁיַּעֲזֹר לוֹ אֶלָּא אִשְׁתּוֹ – מֻתֶּרֶת לְשַׁמְּשׁוֹ, כְּגוֹן לַהֲקִימוֹ אוֹ לְהַשְׁכִּיבוֹ וכיו"ב. וְתִשְׁתַּדֵּל לֹא לִנְגֹּעַ בְּגוּפוֹ אֶלָּא בְּאֶמְצָעוּת מַגֶּבֶת וכיו"ב אוֹ לְפָחוֹת ע"י בֶּגֶד אוֹ כְּפָפוֹת וּלְלֹא מַגָּע בִּבְשָׂרוֹ.

וְכֵן מֻתֶּרֶת לִמְזֹג וּלְהַגִּישׁ לוֹ כּוֹס אוֹ מַאֲכָל, אֲבָל תִּשְׁתַּדֵּל לַעֲשׂוֹת זֹאת בְּאֹפֶן הַמֻּתָּר, כְּגוֹן בְּיַד שְׂמֹאל אוֹ בְּרִחוּק קְצָת וְכַנַּ"ל, אוֹ שֶׁהוּא לֹא יַבִּיט בִּזְמַן הַהַגָּשָׁה, אֲבָל אִם אֵין אֶפְשָׁרוּת אַחֶרֶת – מֻתֶּרֶת בַּכֹּל.

וְתִשְׁתַּדֵּל לְהִמָּנַע מֵרְחִיצַת פָּנָיו יָדָיו וְרַגְלָיו וְהַצָּעַת הַמִּטָּה בְּפָנָיו. וְאִם יֵשׁ לוֹ צֹרֶךְ בָּזֶה – תִּשְׁתַּדֵּל עכ"פ לִרְחֹץ ע"י כְּפָפוֹת וכיו"ב וּלְהַצִּיעַ הַמִּטָּה שֶׁלֹּא בְּפָנָיו אוֹ עכ"פ שֶׁלֹּא יַבִּיט, כַּנַּ"ל.

וְכָל זֶה, כָּאָמוּר, בְּאָדָם חוֹלֶה וְתָשׁוּשׁ, אֲבָל אָדָם שֶׁמְּתַפְקֵד כָּרָגִיל וְיֵשׁ לוֹ רַק כְּאֵב מְקוֹמִי וְכֵן אִם יֵשׁ לוֹ שֶׁבֶר וכד' אֲבָל בִּכְלָלוּת מַרְגִּישׁ טוֹב – דִּינוֹ כְּבָרִיא. וְכֵן אִם יֵשׁ לוֹ מֵחוּשׁ בִּכְלָלוּת הַגּוּף, כְּגוֹן כְּאֵב רֹאשׁ קַל שֶׁנִּתָּן לְסָבְלוֹ וּלְתַפְקֵד כָּרָגִיל אוֹ שֶׁנָּטַל רְפוּאוֹת וְעי"ז נִרְגַּע הַכְּאֵב הֶחָזָק וְיָכוֹל לְתַפְקֵד כָּרָגִיל אוֹ שֶׁיֵּשׁ לוֹ מְעַט חֹם וּמְתַפְקֵד כָּרָגִיל – דִּינוֹ כְּבָרִיא לְעִנְיָן זֶה (וְאִם הַבַּעַל מֻגְבָּל בִּתְנוּעוֹתָיו, כְּגוֹן עֵקֶב פְּצִיעָה וְזָקוּק מְאֹד לְעֶזְרַת הָאִשָּׁה אַךְ אֵינוֹ מֻגְדָּר כְּחוֹלֶה, יֵשׁ לִשְׁאֹל רַב).

בְּמָקוֹם שֶׁמֻּתָּר לָאִשָּׁה לַעֲזֹר לְבַעַל חוֹלֶה – אֵין צְרִיכִים לִשְׂכֹּר בְּכֶסֶף גֶּבֶר שֶׁיְּטַפֵּל בַּבַּעַל, אֲבָל אִם נִתָּן לְבַקֵּשׁ וּלְקַבֵּל עֶזְרָה מִגֶּבֶר לְלֹא תַּשְׁלוּם – צְרִיכָה לְבַקֵּשׁ זֶה גַּם אִם זֶה כָּרוּךְ בְּאִי נְעִימוּת31.

    וְעַל כֵּן אֶפְשָׁר לָקַחַת אָז גַּם זוּג בִּזְמַן שֶׁהָאִשָּׁה טְמֵאָה, כֵּיוָן שֶׁכַּנַּ"ל, בְּכָל מַצָּב אֵין מַעֲבִירִים מִיָּד לְיָד בת"ב וּבְיוהכ"פ. אוֹ שֶׁיִּקְחוּ אָב וּבִתּוֹ אוֹ אָח וַאֲחוֹתוֹ שֶׁמֻּתָּרִים לִמְסֹר מִיָּד לְיָד גַּם בְּיָמִים אֵלּוּ. ↩
