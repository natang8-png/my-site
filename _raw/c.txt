28: בשו"ע לֹא הֻזְכַּר אִסּוּר עַל נְתִינַת מַתָּנָה וּלְדַעַת רֹב הַפּוֹסְקִים מֻתָּר לִתֵּן מַתָּנָה בִּימֵי טֻמְאָתָהּ אַךְ בְּמַתָּנָה חֲשׁוּבָה יֵשׁ לְהַחְמִיר. ¶ וּרְאֵה בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד ס"ק, שֶׁמ"מ יֵשׁ לְהַתִּיר בְּעֵת הַצֹּרֶךְ אִם מְנִיעַת הַמַּתָּנָה עֲלוּלָה לְהָבִיא לִידֵי קְפֵידָא [וּבִפְרָט לְאַחַר לֵדָה יֵשׁ לִתֵּן מַתָּנָה]. וְיֵשׁ שֶׁיָּעֲצוּ לָתֵת לָהּ בֵּנְתַיִם בְּתוֹר הַשְׁאָלָה וְתוּכַל לְהִשְׁתַּמֵּשׁ בַּמַּתָּנָה מִיָּד, אַךְ תִּזְכֶּה בָּהּ בְּפֹעַל רַק אַחַר טָהֳרָתָהּ. וּבְסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פל"ב ס"ה מְחַלֵּק, שֶׁרַק בְּמַתָּנָה חֲשׁוּבָה אוֹ בְּחֵפֶץ שֶׁל חִבָּה יֵשׁ לְהַחְמִיר. ¶ וְשָׁם בְּס"ו מַתִּיר לִשְׁלֹחַ פְּרָחִים אִם רָגִיל בָּזֶה בְּכָל שַׁבָּת וְיו"ט, אֲבָל אִם אֵינוֹ רָגִיל בָּזֶה – יֵשׁ לְהַחְמִיר.

<p><b class="hl">כט.</b> הַבַּעַל וְאִשְׁתּוֹ אֲסוּרִים לְהַצִּיעַ הַמִּטָּה זל"ז בִּזְמַן טֻמְאָתָהּ. אִסּוּר זֶה כּוֹלֵל סִדּוּר הַשְּׂמִיכָה, הַכָּרִים וְסִדּוּר הַסָּדִין [אֲבָל מֻתָּר לְהָבִיא מִזְרָן לְמִטַּת בֶּן הַזּוּג]. וְכֵן מֻתָּרִים לְהַחֲלִיף צִפִּיּוֹת הַכָּרִים וְהַשְּׂמִיכוֹת אֶחָד בִּפְנֵי הַשֵּׁנִי. הָאִסּוּר הוּא רַק הֲכָנַת הַמִּטָּה לְשֵׁנָה אֲבָל אֵין אִסּוּר לְסַדֵּר אֶת הַמִּטָּה לֹא כַּהֲכָנָה לְשֵׁנָה אֶלָּא בְּאֹפֶן שֶׁהַחֶדֶר יֵרָאֶה מְסֻדָּר, אֲבָל לְסַדֵּר לִקְרַאת שֵׁנָה – אָסוּר אֲפִלּוּ אִם מְסַדְּרִים בַּבֹּקֶר עֲבוּר הַלַּיְלָה.</p>
<p>הָאִסּוּר הוּא רַק בִּפְנֵי בֶּן הַזּוּג. וְנֶחְלְקוּ הַפּוֹסְקִים הַאִם הַהֶתֵּר הוּא רַק אִם בֶּן הַזּוּג אֵינוֹ בַּחֶדֶר אוֹ שֶׁמֻּתָּר גַּם אִם הוּא בַּחֶדֶר וּמְסוֹבֵב פָּנָיו שֶׁלֹּא יִרְאֶה [וְאִם הִצִּיעוּ בְּאִסּוּר, הֲרֵי בְּדִיעֲבַד מֻתָּר לִישֹׁן ע"ז (וּרְאֵה בַּהֶעָרָה)]⟦29⟧.</p>

29: רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה סי"א. וּמִסִּגְנוֹן דְּבָרָיו מַשְׁמָע קְצָת, שֶׁאִם הַבַּעַל נִמְצָא – אָסוּר אע"פ שֶׁאֵינוֹ רוֹאֶה, שֶׁהֲרֵי מָצָא הֶתֵּר רַק שֶׁלֹּא בְּפָנָיו, וְלֹא כָּתַב שֶׁיְּסוֹבֵב פָּנָיו וְלֹא יִסְתַּכֵּל. וְאוּלַי בִּשְׁעַת הַדְּחָק נִתָּן לְהַתִּיר כְּשֶׁאֵין הַשֵּׁנִי רוֹאֶה וּכְמ"ש בְּסֵפֶר טָהֳרָה בַּהֲלָכָה פי"ד סק"ב, שֶׁרֹב הַפּוֹסְקִים מַתִּירִים אִם הַשֵּׁנִי אֵינוֹ רוֹאֶה. ¶ וּגְבֵי הַנָּחַת מִזְרָן עַל מִטַּת בֶּן הַזּוּג – אֶפְשָׁר שֶׁבְּמִזְרָן קַל מְאֹד יֵשׁ לֶאֱסֹר. ¶ בְּדִיעֲבַד, אִם הִצִּיעוּ בְּאִסּוּר – מֻתָּר לִישֹׁן עַל זֶה, כְּמ"ש בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סק"ח. ¶ וְיֵשׁ שֶׁהֶחְמִירוּ לִסְתֹּר אֶת סִדּוּר הַמִּטָּה שֶׁנַּעֲשָׂה בְּאִסּוּר, וֶהֱבִיאָם בְּסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פכ"ט סט"ו וּלְדַעְתּוֹ נָכוֹן לְהַחְמִיר בָּזֶה.

<p><b class="hl">ל.</b> אָסוּר לִיצֹק מַיִם כְּדֵי לִרְחֹץ פָּנִים, יָדַיִם וְרַגְלַיִם שֶׁל בֶּן הַזּוּג בִּזְמַן הָאִסּוּר. וְנוֹהֲגִים לֶאֱסֹר גַּם לְהַגִּישׁ קְעָרַת מַיִם לִרְחִיצָה אוֹ לְמַלֵּא אַמְבַּטְיָה מַיִם עֲבוּר הַשֵּׁנִי בְּפָנָיו [וְיֵשׁ שֶׁאָסְרוּ גַּם לְכַוֵּן אֶת זְרִימַת הַמַּיִם בַּמִּקְלַחַת עֲבוּר הַשֵּׁנִי] אֲבָל אִם בֶּן הַזּוּג לֹא נִמְצָא בַּמָּקוֹם – מֻתָּר. וְנֶחְלְקוּ הַאִם מֻתָּר גַּם בְּמִקְרֶה שֶׁהַשֵּׁנִי נִמְצָא בַּמָּקוֹם וְאֵינוֹ מַבִּיט, כַּנַּ"ל הֲלָכָה כט.</p>
<p>וְנֶחְלְקוּ הַפּוֹסְקִים הַאִם בִּכְלַל הָאִסּוּר גַּם הַגָּשַׁת מַיִם לִנְטִילַת יָדַיִם בַּבֹּקֶר אוֹ לִנְטִילַת יָדַיִם לִסְעוּדָה אוֹ לְמַיִם אַחֲרוֹנִים אוֹ לִנְטִילַת יָדַיִם לִפְנֵי תְּפִלָּה, שֶׁיֵּשׁ שֶׁאָסְרוּ גַּם בָּזֶה וְיֵשׁ שֶׁהִתִּירוּ, כִּי אֵין זֶה לְשֵׁם רְחִיצָה [וְכָל זֶה כְּשֶׁאֵין מוֹסְרִים מִיָּד לְיָד אֶלָּא מַנִּיחִים לִפְנֵי הַשֵּׁנִי, כִּי מְסִירָה מִיָּד לְיָד הִיא אִסּוּר וַדַּאי]. וְלָכֵן, לְכַתְּחִלָּה יֵשׁ לְהִזָּהֵר לְהַגִּישׁ כְּשֶׁהַשֵּׁנִי אֵינוֹ רוֹאֶה [אוֹ לְהָבִיא שְׁנֵי סְפָלִים כְּשֶׁאֵין מְבֹרָר מַה עֲבוּר הַבַּעַל וּמַה עֲבוּר הָאִשָּׁה, אוֹ לְמַלֵּא לְעַצְמוֹ וְהַשֵּׁנִי יִטֹּל יָדָיו בַּמַּיִם הַנּוֹתָרִים]⟦30⟧.</p>

30: רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה סי"ב, סֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סקי"ד וְכֵן בְּסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פכ"ז ס"ד. וְשָׁם מְבֹאָר, שֶׁיֵּשׁ מָקוֹם יוֹתֵר לְהָקֵל כְּשֶׁמְּכִינִים לִפְנֵי הַשֵּׁנָה מַיִם לַבֹּקֶר. וּמ"מ מֵבִיא, שֶׁגַּם בָּזֶה יֵשׁ מָקוֹם לְהַחְמִיר לְכַתְּחִלָּה, כַּנַּ"ל בִּפְנִים.
=== 21
<p><b class="hl">לא.</b> אִם הַבַּעַל הוּא חוֹלֶה וְתָשׁוּשׁ, כָּךְ שֶׁאֵין חֲשָׁשׁ שֶׁיָּבוֹא לְתשהמ"ט בְּאִסּוּר, וְאֵין מִי שֶׁיַּעֲזֹר לוֹ אֶלָּא אִשְׁתּוֹ – מֻתֶּרֶת לְשַׁמְּשׁוֹ, כְּגוֹן לַהֲקִימוֹ אוֹ לְהַשְׁכִּיבוֹ וכיו"ב. וְתִשְׁתַּדֵּל לֹא לִנְגֹּעַ בְּגוּפוֹ אֶלָּא בְּאֶמְצָעוּת מַגֶּבֶת וכיו"ב אוֹ לְפָחוֹת ע"י בֶּגֶד אוֹ כְּפָפוֹת וּלְלֹא מַגָּע בִּבְשָׂרוֹ.</p>
<p>וְכֵן מֻתֶּרֶת לִמְזֹג וּלְהַגִּישׁ לוֹ כּוֹס אוֹ מַאֲכָל, אֲבָל תִּשְׁתַּדֵּל לַעֲשׂוֹת זֹאת בְּאֹפֶן הַמֻּתָּר, כְּגוֹן בְּיַד שְׂמֹאל אוֹ בְּרִחוּק קְצָת וְכַנַּ"ל, אוֹ שֶׁהוּא לֹא יַבִּיט בִּזְמַן הַהַגָּשָׁה, אֲבָל אִם אֵין אֶפְשָׁרוּת אַחֶרֶת – מֻתֶּרֶת בַּכֹּל.</p>
<p>וְתִשְׁתַּדֵּל לְהִמָּנַע מֵרְחִיצַת פָּנָיו יָדָיו וְרַגְלָיו וְהַצָּעַת הַמִּטָּה בְּפָנָיו. וְאִם יֵשׁ לוֹ צֹרֶךְ בָּזֶה – תִּשְׁתַּדֵּל עכ"פ לִרְחֹץ ע"י כְּפָפוֹת וכיו"ב וּלְהַצִּיעַ הַמִּטָּה שֶׁלֹּא בְּפָנָיו אוֹ עכ"פ שֶׁלֹּא יַבִּיט, כַּנַּ"ל.</p>
<p>וְכָל זֶה, כָּאָמוּר, בְּאָדָם חוֹלֶה וְתָשׁוּשׁ, אֲבָל אָדָם שֶׁמְּתַפְקֵד כָּרָגִיל וְיֵשׁ לוֹ רַק כְּאֵב מְקוֹמִי וְכֵן אִם יֵשׁ לוֹ שֶׁבֶר וכד' אֲבָל בִּכְלָלוּת מַרְגִּישׁ טוֹב – דִּינוֹ כְּבָרִיא. וְכֵן אִם יֵשׁ לוֹ מֵחוּשׁ בִּכְלָלוּת הַגּוּף, כְּגוֹן כְּאֵב רֹאשׁ קַל שֶׁנִּתָּן לְסָבְלוֹ וּלְתַפְקֵד כָּרָגִיל אוֹ שֶׁנָּטַל רְפוּאוֹת וְעי"ז נִרְגַּע הַכְּאֵב הֶחָזָק וְיָכוֹל לְתַפְקֵד כָּרָגִיל אוֹ שֶׁיֵּשׁ לוֹ מְעַט חֹם וּמְתַפְקֵד כָּרָגִיל – דִּינוֹ כְּבָרִיא לְעִנְיָן זֶה (וְאִם הַבַּעַל מֻגְבָּל בִּתְנוּעוֹתָיו, כְּגוֹן עֵקֶב פְּצִיעָה וְזָקוּק מְאֹד לְעֶזְרַת הָאִשָּׁה אַךְ אֵינוֹ מֻגְדָּר כְּחוֹלֶה, יֵשׁ לִשְׁאֹל רַב).</p>
<p>בְּמָקוֹם שֶׁמֻּתָּר לָאִשָּׁה לַעֲזֹר לְבַעַל חוֹלֶה – אֵין צְרִיכִים לִשְׂכֹּר בְּכֶסֶף גֶּבֶר שֶׁיְּטַפֵּל בַּבַּעַל, אֲבָל אִם נִתָּן לְבַקֵּשׁ וּלְקַבֵּל עֶזְרָה מִגֶּבֶר לְלֹא תַּשְׁלוּם – צְרִיכָה לְבַקֵּשׁ זֶה גַּם אִם זֶה כָּרוּךְ בְּאִי נְעִימוּת⟦31⟧.</p>
<p><b class="hl">לב.</b> אִם הָאִשָּׁה חוֹלָה – אָסוּר לַבַּעַל לְשַׁמְּשָׁהּ אֶלָּא אִם כֵּן זְקוּקָה מְאֹד לְעֶזְרָתוֹ. וְכַמּוּבָן בְּמִקְרֶה שֶׁנִּפְצְעָה אוֹ הִתְעַלְּפָה וּזְקוּקָה מִיָּדִית לְעֶזְרָתוֹ – וַדַּאי שֶׁצָּרִיךְ לְעוֹזְרָהּ.</p>
<p>אֲבָל כְּשֶׁיָּדוּעַ שֶׁתִּזְדַּקֵּק לְסִיּוּעַ בִּזְמַן מְסֻיָּם, עַל הַבַּעַל לִדְאֹג שֶׁאִשָּׁה תְּטַפֵּל בָּהּ גַּם אִם זֶה כָּרוּךְ בְּתַשְׁלוּם כַּסְפִּי וְאִם קָשֶׁה עָלָיו הַהוֹצָאָה הַכַּסְפִּית אוֹ שֶׁיֵּשׁ בָּזֶה קֹשִׁי אַחֵר – יִשְׁאַל רַב⟦32⟧.</p>
<p><b class="hl">לג.</b> אִם הַבַּעַל רוֹפֵא אוֹ אָח רְפוּאִי – לֹא יְטַפֵּל בְּאִשְׁתּוֹ בִּימֵי נִדָּתָהּ, אֲבָל אִם יֵשׁ חֲשַׁשׁ סַכָּנָה וְאֵין שָׁם רוֹפֵא אַחֵר אוֹ שֶׁבַּעְלָהּ בָּקִי יוֹתֵר בָּרְפוּאָה – מֻתָּר [וּמוּבָן שֶׁמֻּתָּר לַבַּעַל לְהַגִּישׁ כָּל סִיּוּעַ דָּחוּף, כְּגוֹן אִם נִתְעַלְּפָה אוֹ נִפְצְעָה, כַּנַּ"ל הֲלָכָה לב]⟦33⟧.</p>

31: רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצד סקט"ו. וְסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סקי"ח וּבְסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פל"ט.
32: רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה סקי"ז. וכ"כ בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סקכ"א וְאֵילָךְ. וְשָׁם מַתִּיר בְּצֹרֶךְ גָּדוֹל לְהוֹלִיכָהּ בְּכִסֵּא גַּלְגַּלִּים.
33: רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה סקי"ז וְשָׁם מְבָאֵר מַדּוּעַ אָסוּר לְבַעַל רוֹפֵא לִבְדֹּק אֶת אִשְׁתּוֹ בִּימֵי טֻמְאָתָהּ, כֵּיוָן שֶׁיֵּשׁ כַּמָּה סְפֵקוֹת אִם יָבוֹא מִזֶּה תּוֹעֶלֶת וְרַק בִּמְקוֹם חֲשַׁשׁ סַכָּנָה יֵשׁ לְהַתִּיר לְבַעְלָהּ רוֹפֵא לְבָדְקָהּ, אִם הוּא מֻמְחֶה יוֹתֵר מֵרוֹפְאִים אֲחֵרִים אוֹ שֶׁאֵין שָׁם רוֹפֵא אַחֵר. וְכֵן הַדִּין אִם הַבַּעַל הוּא "אָח רְפוּאִי" וכיו"ב.
=== 22
<p><b class="hl">לד.</b> נָהֲגוּ הַנָּשִׁים שֶׁלֹּא לֵילֵךְ לְבֵית הַקְּבָרוֹת אוֹ לְהַלְוָיָה בִּימֵי נִדָּתָן וּלְכַתְּחִלָּה נִמְנָעוֹת עַד אַחַר הַטָּהֳרָה בַּמִּקְוֶה וּלְפָחוֹת מַקְפִּידוֹת לְהִמָּנַע בִּזְמַן הַדִּמּוּם, וּמ"מ, בְּצֹרֶךְ גָּדוֹל מַתִּירִים. אֲבָל יֵשׁ לָהּ לְהִשְׁתַּדֵּל לְהַרְחִיק ד' אַמּוֹת מֵהַמִּטָּה אוֹ מֵהַקֶּבֶר. וְכֵן יֵשׁ לָהּ לְהִשְׁתַּדֵּל שֶׁלֹּא לְהִכָּנֵס לְתוֹךְ ד' אַמּוֹת שֶׁל קֶבֶר צַדִּיק וְלֹא לְתוֹךְ אֹהֶל צַדִּיק גַּם בְּרִחוּק ד' אַמּוֹת מִקֶּבֶר צַדִּיק בִּזְמַן הַדִּמּוּם עכ"פ, אַךְ לַכֹּתֶל הַמַּעֲרָבִי נָהֲגוּ לְהָקֵל [כִּי לְרֹב הַפּוֹסְקִים הוּא כֹּתֶל הַר הַבַּיִת וְלֹא כֹּתֶל הָעֲזָרָה].</p>
<p>וְכֵן נָהֲגוּ הַנָּשִׁים בִּזְמַן הַהֵרָיוֹן שֶׁלֹּא לֵילֵךְ לְהַלְוָיָה אוֹ לְבֵית הַקְּבָרוֹת וּלְצֹרֶךְ גָּדוֹל מַתִּירִים רַק עָלֶיהָ לְהִשְׁתַּדֵּל לְהַרְחִיק ד' אַמּוֹת מֵהַקֶּבֶר אוֹ מֵהַמֵּת, אֲבָל אֵין מְנִיעָה בִּזְמַן הַהֵרָיוֹן לְבַקֵּר בְּקִבְרֵי צַדִּיקִים שֶׁאֵינָם בְּבֵית עָלְמִין.</p>
<p>אֵשֶׁת כֹּהֵן מְעֻבֶּרֶת – מִן הַדִּין מֻתֶּרֶת לְהִטַּמֵּא, אַךְ יֵשׁ שֶׁנָּהֲגוּ לְהַחְמִיר בָּזֶה בִּפְרָט כְּשֶׁיָּדוּעַ שֶׁהָעֻבָּר זָכָר. וְכֵן מַחְמִירוֹת שֶׁלֹּא לֵילֵךְ לְקִבְרֵי צַדִּיקִים בִּזְמַן הַהֵרָיוֹן [וְכֵן יֵשׁ לְאֵשֶׁת כֹּהֵן לֵילֵד בְּבֵית־רְפוּאָה שֶׁמֻּחְזָק שֶׁאֵין שָׁם טֻמְאַת מֵת (אִם הוּא בְּרָמָה רְפוּאִית טוֹבָה), כְּדֵי שֶׁלֹּא לְטַמֵּא אֶת הַנּוֹלָד כֹּהֵן בְּטֻמְאַת מֵת. וְכַמּוּבָן בְּכָל מַצָּב דָּחוּף יֵשׁ לִפְנוֹת לְבֵית־הָרְפוּאָה הַקָּרוֹב, וְאִם יֵשׁ צֹרֶךְ, מֻתָּר לְבַעְלָהּ הַכֹּהֵן לֵילֵךְ עִמָּהּ לְשָׁם, אַךְ אִם רוֹצֶה לִשְׁהוֹת שָׁם אוֹ לְבַקְּרָהּ שָׁם – יִשְׁאַל רַב]⟦34⟧.</p>

34: כ"כ בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סעפים קכט-קל. ¶ וּבְסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פמ"ב ס"ח מֵבִיא שֶׁרָאוּי שֶׁלֹּא לֵילֵךְ לְקֶבֶר רשב"י [וְכָל הַצַּדִּיקִים בִּזְמַן הַדִּמּוּם, אַךְ בַּבָּאָה מִדֶּרֶךְ רְחוֹקָה מְקִלִּים. וְיֵשׁ שֶׁאֵין נִכְנָסוֹת לְתוֹךְ אֹהֶל רשב"י אֶלָּא עוֹמְדוֹת בַּחוּץ, וְכֵן בִּשְׁאָר הַצַּדִּיקִים שֶׁיֵּשׁ לָהֶם אֹהֶל עִם גַּג עַל הַקֶּבֶר יֵשׁ שֶׁמַּחְמִירוֹת שֶׁאַף אִם בָּאוּ מִמֶּרְחָק מַמְתִּינוֹת בַּחוּץ].
=== 23
<p><b class="hl">לה.</b> נָהֲגוּ הַנָּשִׁים שֶׁלֹּא לֵילֵךְ לְבֵית־הַכְּנֶסֶת בִּזְמַן הַדִּמּוּם [אֲבָל כְּשֶׁפָּסַק הַדִּמּוּם אע"פ שֶׁעֲדַיִן לֹא עָשְׂתָה הֶפְסֵק טָהֳרָה – מֻתֶּרֶת לֵילֵךְ לְבֵית־הַכְּנֶסֶת].</p>
<p>וְאִם נוֹהֶגֶת לָלֶכֶת בְּכָל שַׁבָּת לְבֵית־הַכְּנֶסֶת אוֹ שֶׁיֵּשׁ לָהּ סִבָּה חֲזָקָה לֵילֵךְ לְבֵית־הַכְּנֶסֶת, כְּגוֹן בְּיָמִים נוֹרָאִים, חַגִּים, אוֹ שִׂמְחָה הַנֶּעֱרֶכֶת בְּבֵית־הַכְּנֶסֶת שֶׁצְּרִיכָה לְהִשְׁתַּתֵּף בָּהּ – מַתִּירִים לָהּ לָלֶכֶת לְבֵית־הַכְּנֶסֶת. וּמ"מ, תִּזָּהֵרְנָה שֶׁלֹּא לִרְאוֹת סֵפֶר תּוֹרָה בִּזְמַן הַהַגְבָּהָה [כְּשֶׁפּוֹתְחִין הַכְּתָב].</p>
<p>וְכֵן מֻתֶּרֶת לְהִשְׁתַּתֵּף בְּשִׁעוּר תּוֹרָה, הַרְצָאָה אוֹ הִתְוַעֲדוּת לְנָשִׁים. וּבְוַדַּאי שֶׁמֻּתָּר לַמַּרְצֶה וכיו"ב לָבוֹא לְבֵית־הַכְּנֶסֶת לְהַרְצוֹת לְנָשִׁים גַּם אִם הַהַרְצָאָה נֶעֱרֶכֶת בְּבֵית־הַכְּנֶסֶת עַצְמוֹ וְוַדַּאי מְקִלִּים אִם נֶעֱרֶכֶת בְּעֶזְרַת נָשִׁים.</p>
<p>וְכֵן יֵשׁ לְהַתִּיר לְהַכְנִיס נְשֵׁי וּבְנוֹת יִשְׂרָאֵל לְבֵית־הַכְּנֶסֶת לְצֹרֶךְ הַרְצָאָה וכיו"ב כְּדֵי לְקָרְבָן לִשְׁמִירַת תּוֹרָה וּמִצְווֹת⟦35⟧.</p>

37: רְאֵה שוע"ר יו"ד סי' קפד סק"ו, וּבְאו"ח סי' תרטו ס"א, שֶׁיֵּשׁ לְהַחְמִיר בְּיוהכ"פ בְּכָל הַהַרְחָקוֹת בַּלַּיְלָה וּבַיּוֹם. ¶ וְשָׁם בְּסִי' מ קו"א סק"ב, שֶׁגַּם גַּבֵּי ט' בְּאָב אֵין לְחַלֵּק בֵּין לַיְלָה לְיוֹם [וְזוֹ לְשׁוֹן רַבֵּנוּ שָׁם בְּס"ח בַּמֻּסְגָּר: "שֶׁהֲרֵי מִן הַדִּין מֻתָּר לִישֹׁן עִם אִשְׁתּוֹ בת"ב וְאֵין חוֹשְׁשִׁין שֶׁמָּא יִשְׁכַּח (וִישַׁמֵּשׁ), אֶלָּא שֶׁנָּכוֹן הַדָּבָר לְהַחְמִיר"]. ¶ וְעַל כֵּן, אִם יֵשׁ בְּרִית מִילָה בת"ב אוֹ בְּיוהכ"פ, יֵשׁ לְהִזָּהֵר שֶׁה"קוואטער" [הַזּוּג שֶׁמַּעֲבִירִים אֶת הַיֶּלֶד הַנִּמּוֹל מֵאִמּוֹ לְאָבִיו כְּדֵי לְמוּלוֹ] לֹא יִמְסְרוּ מִיָּד לְיָד אֶלָּא יַנִּיחוּהוּ בֵּינֵיהֶם עַל אֵיזֶה מָקוֹם, וְאָז הַשֵּׁנִי יִטְּלֶנּוּ (רְאֵה הֶעָרָה 8). ¶ וְעַל כֵּן אֶפְשָׁר לָקַחַת אָז גַּם זוּג בִּזְמַן שֶׁהָאִשָּׁה טְמֵאָה, כֵּיוָן שֶׁכַּנַּ"ל, בְּכָל מַצָּב אֵין מַעֲבִירִים מִיָּד לְיָד בת"ב וּבְיוהכ"פ. אוֹ שֶׁיִּקְחוּ אָב וּבִתּוֹ אוֹ אָח וַאֲחוֹתוֹ שֶׁמֻּתָּרִים לִמְסֹר מִיָּד לְיָד גַּם בְּיָמִים אֵלּוּ.
