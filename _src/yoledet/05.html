ז. יוֹלֶדֶת נֶחְשֶׁבֶת כִּמְסֻכֶּנֶת בִּשְׁלֹשָׁה הַיָּמִים הָרִאשׁוֹנִים אַחַר הַלֵּדָה, וּמְחַלְּלִים שַׁבָּת בִּשְׁבִילָהּ. וְלָכֵן כָּל דָּבָר שֶׁזְּקוּקָה לוֹ [כְּגוֹן מָזוֹן חַם] וְלֹא נִתָּן לְהַשִּׂיגוֹ בְּלִי שֶׁיְּחַלְּלוּ שַׁבָּת, יֵשׁ לְחַלֵּל שַׁבָּת בִּשְׁבִילָהּ [אַךְ בִּזְמַנֵּנוּ ע"פ הָרֹב אֵין הַיּוֹלֶדֶת זְקוּקָה לְמָזוֹן חַם בְּדַוְקָא].

כֵּן אֲסוּרָה הַיּוֹלֶדֶת לָצוּם בְּיוהכ"פ, וכ"ש בִּשְׁאָר יָמִים, בְּתוֹךְ ג' יָמִים מֵהַלֵּדָה.

וּמִשְּׁלֹשָׁה עַד שִׁבְעָה יָמִים מֵהַלֵּדָה, אִם אָמְרָה "צְרִיכָה אֲנִי מַאֲכָל חַם" וכיו"ב, מְחַלְּלִים שַׁבָּת עֲבוּרָהּ.

וְכֵן לְעִנְיַן יוהכ"פ, אִם אָמְרָה שֶׁזְּקוּקָה לַאֲכִילָה, תֹּאכַל. מ"מ, כָּל יוֹלֶדֶת שֶׁאוֹכֶלֶת בְּיוהכ"פ תֹּאכַל לְשִׁעוּרִין, אִם אֵינָהּ זְקוּקָה מִיָּדִית לְכַמּוּת גְּדוֹלָה שֶׁל מַאֲכָל אוֹ מַשְׁקֶה.

וּמִשִּׁבְעָה עַד שְׁלֹשִׁים יוֹם לַלֵּדָה נֶחְשֶׁבֶת כְּחוֹלֶה שֶׁנָּפַל לְמִשְׁכָּב וְאֵין בּוֹ סַכָּנָה, וּפְטוּרָה מִלָּצוּם בִּשְׁאָר תַּעֲנִיּוֹת, אֲבָל בְּתַעֲנִית יוהכ"פ חַיֶּבֶת. וְאִם מַרְגִּישָׁה חֻלְשָׁה, תִּשְׁאַל רַב, וְאִם חוֹשֶׁשֶׁת לְסַכָּנַת נֶפֶשׁ, תֹּאכַל מִיָּד אֲפִלּוּ בִּסְפֵק סַכָּנָה. וְכֵן אִם אָמַר לָהּ רוֹפֵא שֶׁצְּרִיכָה לֶאֱכֹל, צְרִיכָה לֶאֱכֹל. אִם לִפְנֵי יוהכ"פ הָרוֹפֵא אָמַר לָהּ שֶׁצְּרִיכָה לֶאֱכֹל, תִּתְיַעֵץ עִם רַב לִפְנֵי יוהכ"פ כֵּיצַד (הַאִם צְרִיכָה לֶאֱכֹל בְּשִׁעוּרִים וְכוּ').

יוֹלֶדֶת בְּנִתּוּחַ קֵיסָרִי אֶפְשָׁר שֶׁנֶּחְשֶׁבֶת לִמְסֻכֶּנֶת יוֹתֵר מֵאֲשֶׁר יוֹלֶדֶת רְגִילָה, ע"כ אִם יָלְדָה לִפְנֵי שַׁבָּת אוֹ יוהכ"פ, תִּתְיַעֵץ לִפְנֵי יוהכ"פ עִם רוֹפֵא וְעִם רַב. וְאִם יָלְדָה בְּשַׁבָּת אוֹ יוהכ"פ, וַדַּאי בְּאוֹתוֹ יוֹם דִּינָהּ כִּמְסֻכֶּנֶת. וְכֵן אִשָּׁה שֶׁאֲחָזוּהָ חֲבָלֵי לֵדָה נֶחְשֶׁבֶת מִיָּד כִּמְסֻכֶּנֶת7.

  1. רְאֵה שוע"ר או"ח סי' של סעיפים ד וה.

    וּבְס"ו שָׁם כּוֹתֵב רַבֵּנוּ, שֶׁמִּקְצָת יוֹם נֶחְשָׁב יוֹם וְלֹא מוֹנִים מֵעֵת לְעֵת. לְדֻגְמָא: אִם יָלְדָה בְּיוֹם ד' בְּשָׁעָה 2 בַּצָּהֳרַיִם, מוֹנִים לָהּ יוֹם ד' וה' וְיוֹם עֶרֶב שַׁבָּת, אֲבָל בְּשַׁבָּת דִּינָהּ כְּמוֹ מג' עַד ז' יָמִים לַלֵּדָה.

    ועד"ז בְּהִלְכוֹת יוהכ"פ בְּסִי' תריז סעיפים ד וה. וְשָׁם נִכְתַּב, שֶׁמַּאֲכִילִים אוֹתָהּ [גַּם] בְּעַל כָּרְחָהּ בְּג' יָמִים רִאשׁוֹנִים [וְשָׁמַעְתִּי שֶׁהָיָה מַעֲשֶׂה בא"י, שֶׁיּוֹלֶדֶת בַּיּוֹם הַשֵּׁנִי לְלֵדָתָהּ צָמָה בְּיוהכ"פ וְהָלְכָה לְעוֹלָמָהּ, ה"י].

    וּבְשו"ת צֶמַח צֶדֶק או"ח סי' סג מֵבִיא מֵאָח אדמוה"ז, בַּעַל שְׁאֵרִית יְהוּדָה, שֶׁפָּסַק לְהָקֵל בְּאִשָּׁה חֲלוּשָׁה לֶאֱכֹל לְשִׁעוּרִין בְּיוהכ"פ עַד שְׁלֹשָׁה יָמִים מֵעֵת לְעֵת.

    לְדֻגְמָא, אִם יָלְדָה בְּיוֹם רְבִיעִי, ז' תִּשְׁרֵי, בְּשָׁעָה 2 בַּצָּהֳרַיִם, נֶחְשֶׁבֶת מְסֻכֶּנֶת עַד יוֹם שַׁבָּת, יוהכ"פ, בְּשָׁעָה 2 בַּצָּהֳרַיִם, וּכְדַעַת הַרְבֵּה פּוֹסְקִים. עי"ש.

    ועד"ז פָּסַק בְּשו"ת שְׁאֵרִית יְהוּדָה חֵלֶק או"ח סי' טו.
