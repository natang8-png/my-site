ד. אַחַר לֵידַת בֵּן עָלֶיהָ לְהַמְתִּין 5 יָמִים מִתְּחִילַּת הָאִיסּוּר וְלִמְנוֹת 7 יָמִים נְקִיִּים אַחַר הַלֵּידָה; וּבְלֵידַת בַּת עָלֶיהָ לְהַמְתִּין 7 יָמִים אַחַר הַלֵּידָה לִפְנֵי סְפִירַת 7 יָמִים נְקִיִּים [אַךְ בְּדֶרֶךְ כְּלָל לוֹקֵחַ זְמַן מְמוּשָּׁךְ עַד שֶׁפּוֹסֵק הַדָּם וְרַק אָז פּוֹסֶקֶת בְּטָהֳרָה, סוֹפֶרֶת 7 יָמִים וְטוֹבֶלֶת].

ה. אִשָּׁה שֶׁהִפִּילָה, ח"ו, אִם זוֹ הָיְיתָה נְקֵבָה אוֹ אִם אֵין יָדוּעַ אִם הַנֵּפֶל הָיָה בֵּן אוֹ בַּת - עָלֶיהָ לְהַמְתִּין 7 יָמִים אַחַר הַהַפָּלָה וְרַק אח"כ לִסְפּוֹר שִׁבְעָה יָמִים נְקִיִּים.

וְאִם יוֹדַעַת שֶׁהַנֵּפֶל הָיָה זָכָר - מַמְתִּינָה חֲמִישָּׁה יָמִים מִתְּחִילַּת טוּמְאָתָהּ ואח"כ סוֹפֶרֶת שִׁבְעָה יָמִים נְקִיִּים.

וְאִם הַהַפָּלָה הָיְיתָה בַּשָּׁבוּעוֹת הָרִאשׁוֹנִים - תִּשְׁאַל רַב הַאִם יְכוֹלָה לְהִסְתַּפֵּק בְּהַמְתָּנַת חֲמִישָּׁה יָמִים ואח"כ שִׁבְעָה יָמִים נְקִיִּים.
