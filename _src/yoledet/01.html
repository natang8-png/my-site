בקיצור

הַיּוֹלֶדֶת (אוֹ מַפֶּלֶת ח"ו) טְמֵאָה כְּנִדָּה.

יוֹלֶדֶת אֵינָהּ טוֹבֶלֶת עַד שֶׁתִּסְפֹּר שִׁבְעָה נְקִיִּים, לְאַחַר שֶׁפָּסַק הַדִּמּוּם. בְּמִקְרֶה וְהַדִּמּוּם פָּסַק מִיָּד (נָדִיר): בְּיוֹלֶדֶת זָכָר, מִינִימוּם הַזְּמַן לְטֻמְאָה הוּא כָּרָגִיל, הַמְתָּנַת חֲמִשָּׁה יָמִים מִתְּחִלַּת הַטֻּמְאָה וְעוֹד שִׁבְעָה נְקִיִּים אַחַר הַלֵּדָה, וְאָז הִיא טוֹבֶלֶת. אֲבָל בְּיוֹלֶדֶת נְקֵבָה לֹא תִּטְבֹּל לִפְנֵי שֶׁעָבְרוּ שְׁבוּעַיִם מֵהַלֵּדָה.

בְּנוֹסָף, בְּיָמֵינוּ מְטַמְּאִים כָּל דָּם שֶׁרוֹאָה יוֹלֶדֶת, גַּם כְּשֶׁהוּא בָּא בִּימֵי הַטֹּהַר. הַסִּבָּה הִיא מִנְהָג שֶׁלֹּא לְחַלֵּק בֵּין דָּם לְדָם, כְּדֵי שֶׁלֹּא יָקֵלּוּ בְּדַם טָמֵא (וְיֵשׁ סוֹבְרִים שֶׁזֶּה כָּלוּל בְּמִנְהַג בְּנוֹת יִשְׂרָאֵל לָשֶׁבֶת שִׁבְעָה נְקִיִּים עַל טִפַּת דָּם, וּלְדַעְתָּם גַּם דַּם טֹהַר בִּכְלַל זֶה).

נֶאֱמַר בַּתּוֹרָה בִּתְחִלַּת פָּרָשַׁת תַּזְרִיעַ, שֶׁאִשָּׁה שֶׁיָּלְדָה זָכָר טְמֵאָה שִׁבְעָה יָמִים וְאח"כ יֵשׁ לָהּ שְׁלֹשִׁים וּשְׁלֹשָׁה יְמֵי טָהֳרָה, וְאִשָּׁה שֶׁיָּלְדָה נְקֵבָה טְמֵאָה אַרְבָּעָה עָשָׂר יוֹם וְאח"כ יֵשׁ לָהּ שִׁשִּׁים וְשִׁשָּׁה יְמֵי טָהֳרָה [וּבְתֹם טֻמְאָתָהּ וִימֵי טָהֳרָתָהּ הָיְתָה מְבִיאָה קָרְבָּנוֹת בִּזְמַן שֶׁבֵּיהמ"ק הָיָה קַיָּם].

כְּלוֹמַר, מִדְּאוֹרַיְתָא יוֹלֶדֶת זָכָר טְמֵאָה שָׁבוּעַ וְאח"כ יֵשׁ לָהּ שְׁלֹשִׁים וּשְׁלֹשָׁה יוֹם שֶׁאֵין דָּם שֶׁלָּהּ מְטַמְּאָהּ. וְלָכֵן אַחַר שָׁבוּעַ הִיא יְכוֹלָה לִטְבֹּל גַּם אִם מְדַמֶּמֶת עֲדַיִן, וְאֵין לָהּ כָּל חֲשָׁשׁ, כִּי הַדָּם הוּא טָהוֹר לְמֶשֶׁךְ שְׁלֹשִׁים וּשְׁלֹשָׁה יָמִים אַחַר הַשָּׁבוּעַ.

יוֹלֶדֶת נְקֵבָה הִיא טְמֵאָה שְׁבוּעַיִם וְאח"כ יֵשׁ לָהּ שִׁשִּׁים וְשִׁשָּׁה יְמֵי טָהֳרָה, וכנ"ל, שֶׁהִיא טוֹבֶלֶת אַחַר שְׁבוּעַיִם, וְגַם אִם רוֹאָה דָּם אֵין כָּל חֲשָׁשׁ, כִּי הַתּוֹרָה טִהֲרָה אֶת הַדָּם הַזֶּה.

אַךְ לְמַעֲשֶׂה, הַיּוֹם מְטַמְּאִים כָּל דָּם שֶׁתִּרְאֶה הָאִשָּׁה גַּם אִם הוּא בִּימֵי טֹהַר, מִצַּד הַמִּנְהָג שֶׁנָּהֲגוּ שֶׁלֹּא לְחַלֵּק בֵּין דָּם לְדָם, כְּדֵי שֶׁלֹּא יָקֵלּוּ בְּדַם טָמֵא [וְיֵשׁ דֵּעוֹת שֶׁזֶּה כָּלוּל בְּמִנְהַג בְּנוֹת יִשְׂרָאֵל לָשֶׁבֶת שִׁבְעָה יָמִים נְקִיִּים עַל טִפַּת דָּם, וּלְדַעְתָּם גַּם דַּם טֹהַר כָּלוּל בַּמִּנְהָג הַזֶּה]. עַל כֵּן אֵין יְכוֹלָה הַיּוֹלֶדֶת לִטְבֹּל עַד שֶׁתִּסְפֹּר שִׁבְעָה נְקִיִּים, דְּהַיְנוּ, בְּיוֹלֶדֶת זָכָר הַמִּינִימוּם הוּא חֲמִשָּׁה יָמִים שֶׁעָלֶיהָ לְהַמְתִּין מִתְּחִלַּת טֻמְאָתָהּ וְעוֹד שִׁבְעָה נְקִיִּים אַחַר הַלֵּדָה, וְתִטְבֹּל. יוֹלֶדֶת נְקֵבָה לֹא תִּטְבֹּל לִפְנֵי שֶׁעָבְרוּ שְׁבוּעַיִם מֵהַלֵּדָה.

בְּטַעַם הַדָּבָר שֶׁיּוֹלֶדֶת נְקֵבָה יְמֵי טֻמְאָתָהּ [מִן הַתּוֹרָה] וִימֵי טָהֳרָתָהּ מְרֻבִּים יוֹתֵר, נֶאֶמְרוּ כַּמָּה הֶסְבֵּרִים, וּבֵינֵיהֶם, שֶׁבְּאֹפֶן טִבְעִי לֵדַת נְקֵבָה מְרֻבָּה בְּדָמִים יוֹתֵר מִבְּלֵדַת זָכָר, וְע"כ גַּם מְרֻבִּים יְמֵי טֻמְאָתָהּ בַּזְּמַן הַסָּמוּךְ לַלֵּדָה, וְכֵן מְרֻבִּים יְמֵי טֹהַר שֶׁלָּהּ אח"כ.

וְהִנֵּה רמ"א בְּשו"ע (סי' קצד ס"א) מֵבִיא, שֶׁיֵּשׁ מְקוֹמוֹת שֶׁנָּהֲגוּ שֶׁאִשָּׁה יוֹלֶדֶת זָכָר טְמֵאָה אַרְבָּעִים יוֹם לְפָחוֹת, וְיוֹלֶדֶת נְקֵבָה טְמֵאָה שְׁמוֹנִים יוֹם לְפָחוֹת. ואע"פ שֶׁלִּכְאוֹרָה זֶה הָפוּךְ מֵהַכָּתוּב בַּתּוֹרָה, שֶׁהֲרֵי בַּתּוֹרָה נִכְתַּב בְּפֵרוּשׁ שֶׁיֵּשׁ לַיּוֹלֶדֶת יְמֵי טֹהַר, מ"מ כָּךְ נָהֲגוּ בְּכַמָּה מְקוֹמוֹת, וְכַמָּה הֶסְבֵּרִים נֶאֶמְרוּ בַּדָּבָר. אֶחָד הַהֶסְבֵּרִים הוּא, דְּהִנֵּה הַתּוֹרָה טִהֲרָה אֶת הַדָּמִים בִּימֵי טָהֳרָהּ, וְהַטַּעַם שֶׁטִּהֲרָה הַתּוֹרָה, כֵּיוָן שֶׁזֶּהוּ זְמַן שֶׁצְּפוּיָה לְדַמֵּם בּוֹ מִצַּד הַלֵּדָה, וְהַתּוֹרָה טִהֲרַתּוּ מֵאַחַר שֶׁיֵּשׁ לָזֶה סִבָּה מְיֻחֶדֶת וְשׁוֹנָה מִשְּׁאָר דַּם הָאִשָּׁה.

וְהִנֵּה, הַיּוֹם נָהֲגוּ לְטַמֵּא כָּל דָּם [כְּדֵי שֶׁלֹּא יָבוֹאוּ לְזַלְזֵל בְּדָם שֶׁמְּטַמֵּא], וְע"כ חָשְׁשׁוּ וְאָסְרוּ אֶת הָאִשָּׁה בְּאוֹתָם יָמִים, מִפְּנֵי שֶׁכָּאָמוּר, הַתּוֹרָה הֵעִידָה שֶׁמָּצוּי אֶצְלָהּ דָּם, וְחָשְׁשׁוּ שֶׁמָּא תִּרְאֶה דָּם וְלֹא תִּהְיֶה מוּדַעַת לְכָךְ, כֵּיוָן שֶׁבְּיָמִים אֵלּוּ מָצוּי אֶצְלָהּ דָּם, כנ"ל.

אוּלָם מִנְהָג זֶה אֵינוֹ נָפוֹץ בִּקְהִלּוֹת הָאַשְׁכְּנַזִּים וְגַם לֹא בְּקֶרֶב אנ"ש, אִם כִּי יֵשׁ קְצָת מִקְּהִלּוֹת הַסְּפָרַדִּים שֶׁנָּהֲגוּ כֵּן.

וְלָכֵן, אִם אִשָּׁה נָהֲגָה כֵּן אוֹ שֶׁאִמָּהּ נָהֲגָה כֵּן, תִּשְׁאַל רַב הַאִם עָלֶיהָ לְהַמְשִׁיךְ בַּמִּנְהָג אוֹ לְבַטְּלוֹ, וְאִם הִיא רוֹצָה שֶׁלֹּא לִנְהֹג כֵּן, הַאִם צְרִיכָה [הִיא וְגַם בַּעְלָהּ] הַתָּרַת נְדָרִים.

מאימתי נטמאת היולדת

שְׁאֵלָה נוֹסֶפֶת שֶׁדָּנוּ בָּהּ הַפּוֹסְקִים הִיא מֵאֵיזֶה זְמַן אִשָּׁה נִטְמֵאת בְּתַהֲלִיךְ הַלֵּדָה. מוּבָן שֶׁמִּזְּמַן שֶׁרָאֲתָה דָּם הִיא נִטְמֵאת, אֲבָל יֵשׁ חֵלֶק מֵהַפּוֹסְקִים הַסּוֹבְרִים שֶׁנִּטְמֵאת כְּבָר מִזְּמַן שֶׁאֵינָהּ יְכוֹלָה לַעֲמֹד עַל רַגְלֶיהָ מֵחֲמַת צִירֵי לֵדָה. וְכֵן נוֹהֲגִים בְּקֶרֶב אנ"ש. וְיֵשׁ שֶׁמְּטַהֲרִים בְּצִירֵי לֵדָה כָּל זְמַן שֶׁאֵין דָּם.

(כֵּן יֵשׁ שֶׁחָשְׁשׁוּ לְטֻמְאָה בְּעֵת יְרִידַת מַיִם בְּכַמּוּת גְּדוֹלָה, שֶׁגַּם אִם לֹא נִרְאֵית בָּהֶם דָּם חָשְׁשׁוּ שֶׁמָּא הִתְעָרֵב שָׁם מְעַט דָּם, וְלָכֵן אִם הָיָה יְרִידַת מַיִם בְּכַמּוּת נִכֶּרֶת, יֵשׁ לִשְׁאֹל רַב).

וְכֵן יֵשׁ לִשְׁאֹל רַב עַל פְּתִיחַת הָרֶחֶם שֶׁנִּגְרְמָה כְּתוֹצָאָה מֵהַצִּירִים, אֲבָל פְּתִיחָה שֶׁקַּיֶּמֶת זְמַן אָרֹךְ בְּמַהֲלַךְ הַהֵרָיוֹן וְלֹא נוֹצְרָה מֵהַצִּירִים, אֵין בָּהּ כָּל חֲשַׁשׁ טֻמְאָה.

כֵּן יֵשׁ לַחְשֹׁשׁ לְטֻמְאָה בְּהַכְנָסַת יַד הָרוֹפֵא בְּסָמוּךְ לַלֵּדָה, כֵּיוָן שֶׁלְּעִתִּים הַכְנָסַת הַיָּד הִיא לְתוֹךְ הָרֶחֶם [כְּדֵי לִגְרֹם זֵרוּז הַלֵּדָה וְכַדּוֹמֶה] וְיֵשׁ בָּזֶה חֲשַׁשׁ טֻמְאָה. וְכֵן יֵשׁ חֲשַׁשׁ טֻמְאָה בְּהַכְנָסַת מַכְשִׁיר לַגּוּף בְּסָמוּךְ לַלֵּדָה [וְלָכֵן בְּכָל סָפֵק תִּשְׁאַל רַב].

הפלה

הַפָּלָה, רח"ל, נֶחְשֶׁבֶת לְלֵדָה, וּבִתְנַאי שֶׁהַהַפָּלָה הָיְתָה בְּהֵרָיוֹן שֶׁנִּמְשַׁךְ לְפָחוֹת אַרְבָּעִים יוֹם, שֶׁבָּהֶם הָיְתָה הִתְפַּתְּחוּת הָעֻבָּר. אֲשֶׁר לָכֵן, אִם הִפִּילָה אַחַר אַרְבָּעִים יוֹם וְלֹא נוֹדַע מִין הָעֻבָּר, עָלֶיהָ לַחְשֹׁשׁ שֶׁמָּא נְקֵבָה הִפִּילָה, וְלָכֵן לֹא תִּטְבֹּל לִפְנֵי שֶׁיַּחְלְפוּ שְׁבוּעַיִם מִלֵּדָתָהּ.

וְאִם הִפִּילָה גּוּשׁ לִפְנֵי אַרְבָּעִים יוֹם, טְמֵאָה כְּנִדָּה, וְתַמְתִּין חֲמִשָּׁה יָמִים וְאח"כ תִּסְפֹּר שִׁבְעָה נְקִיִּים [כִּי אֵין פְּתִיחַת הָרֶחֶם בְּלֹא דָּם].

ניתוח קיסרי

נִתּוּחַ קֵיסָרִי אֵינוֹ נֶחְשָׁב כְּלֵדָה לְעִנְיַן טֻמְאָה. וְלָכֵן, אִם בְּתַהֲלִיךְ הַנִּתּוּחַ לֹא יָצָא דָּם מֵהַמָּקוֹם הָרָגִיל, וְרַק בַּחֲתַךְ הַנִּתּוּחַ יָצָא דָּם, הִיא טְהוֹרָה [אַךְ אֵינוֹ מָצוּי כְּלָל]. אֲבָל אִם יָצָא גַּם דָּם בַּדֶּרֶךְ הָרְגִילָה הִיא טְמֵאָה, אֲבָל גַּם אִם יָלְדָה נְקֵבָה אֵינָהּ צְרִיכָה לְהַמְתִּין שְׁבוּעַיִם עַד הַטְּבִילָה, אֶלָּא מוֹנָה חֲמִשָּׁה יָמִים מִתְּחִלַּת טֻמְאָתָהּ וְסוֹפֶרֶת שִׁבְעָה נְקִיִּים וְטוֹבֶלֶת [אִם אָכֵן מִתְאַפְשֵׁר הַדָּבָר מִצַּד הַמְּצִיאוּת וּמִצַּד בְּרִיאוּת גּוּפָהּ].
