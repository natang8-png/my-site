א. יוֹלֶדֶת אוֹ מַפֶּלֶת עֻבָּר בֶּן אַרְבָּעִים יוֹם וָיוֹתֵר נִטְמֵאת גַּם אִם לֹא רָאֲתָה דָּם בְּתַהֲלִיךְ הַלֵּדָה אוֹ הַהַפָּלָה.

וְאִם הִפִּילָה נֵפֶל שֶׁגִּילוֹ פָּחוֹת מֵאַרְבָּעִים יוֹם, גַּם אִם לֹא הָיָה דִּמּוּם, עָלֶיהָ לִשְׁאֹל רַב, כִּי אִם הֻפַּל גּוּשׁ בְּגֹדֶל מְסֻיָּם יֵשׁ לָהּ טֻמְאָה כְּנִדָּה1.

ב. אִשָּׁה שֶׁאֲחָזוּהָ חֲבָלֵי לֵדָה חֲזָקִים וְאֵינָהּ יְכוֹלָה לַעֲמֹד עַל רַגְלֶיהָ מִגֹּדֶל הַכְּאֵבִים, אוֹ שֶׁמִּסִּבָּה אַחֶרֶת עָלֶיהָ לִשְׁכַּב בְּמִשְׁכַּב הַיּוֹלֶדֶת, הֲרֵי הִיא טְמֵאָה [מִסָּפֵק] גַּם אִם לֹא יָצָא דָּם. וְלָכֵן אֵין לַבַּעַל לִהְיוֹת נוֹכֵחַ בַּחֲדַר הַלֵּדָה, וּלְדַעַת הָרַבִּי עָלָיו לְהִמָּנַע מִלִּהְיוֹת נוֹכֵחַ גַּם בַּחֲדַר הַצִּירִים [כַּנִּרְאֶה מִזְּמַן שֶׁלּוֹבֶשֶׁת חָלוּק לִקְרַאת הַלֵּדָה, אַךְ בִּמְקוֹמוֹת שֶׁלּוֹבֶשֶׁת חָלוּק סָמוּךְ מַמָּשׁ לַלֵּדָה יִתָּכֵן שֶׁעַל הַבַּעַל לָצֵאת גַּם לִפְנֵי שֶׁלּוֹבֶשֶׁת חָלוּק]. אִם פָּסְקוּ חֲבָלֵי הַלֵּדָה, תִּשְׁאַל רַב מַה דִּינָהּ.

אִם הָיְתָה לָהּ יְרִידַת מַיִם בְּכַמּוּת קְטַנָּה לְלֹא דָּם, הִיא טְהוֹרָה, וְאִם הָיְתָה לָהּ יְרִידַת מַיִם בְּכַמּוּת גְּדוֹלָה לְלֹא דָּם, תִּשְׁאַל רַב.

אִם נִבְדְּקָה בְּדִיקָה פְּנִימִית סָמוּךְ לַלֵּדָה, כְּגוֹן מִתּוֹךְ רָצוֹן לְזָרֵז אֶת הַלֵּדָה אוֹ מִפְּנֵי צֹרֶךְ לִבְדֹּק בְּתוֹךְ הָרֶחֶם, תִּשְׁאַל רַב. אֲבָל פְּתִיחָה מְסֻיֶּמֶת שֶׁיֵּשׁ לָהּ זְמַן נִכָּר לִפְנֵי הַלֵּדָה, אֵין בָּהּ חֲשָׁשׁ וּטְהוֹרָה הִיא2.

ג. כֵּיצַד סֵדֶר טָהֳרַת הַיּוֹלֶדֶת? יוֹלֶדֶת זָכָר, מוֹנָה חֲמִשָּׁה יָמִים מִתְּחִלַּת הַטֻּמְאָה וְסוֹפֶרֶת שִׁבְעָה נְקִיִּים אַחַר הַלֵּדָה וְטוֹבֶלֶת. יוֹלֶדֶת נְקֵבָה, לֹא תִּטְבֹּל לִפְנֵי שֶׁעָבְרוּ אַרְבָּעָה עָשָׂר יָמִים מִלֵּדָתָהּ3.

  1. רְאֵה שו"ע סי' קצד סעיפים א וב. ↩

  2. נֶחְלְקוּ הַפּוֹסְקִים הַאִם הַכְּלָל שֶׁנֶּאֱמַר בְּשַׁבָּת [רְאֵה שוע"ר סי' של ס"ג], שֶׁנִּקְרֵאת יוֹלֶדֶת לְחִלּוּל שַׁבָּת מִשֶּׁתֵּשֵׁב עַל הַמַּשְׁבֵּר אוֹ שֶׁאֵין לָהּ כֹּחַ לָלֶכֶת, הַאִם זֶה גַּם נוֹגֵעַ לַחֲשַׁשׁ טֻמְאָה, אוֹ שֶׁרַק לְעִנְיַן שַׁבָּת חוֹשְׁשִׁין לְפִקּוּחַ נֶפֶשׁ, אֲבָל אֵינָהּ טְמֵאָה עֲדַיִן עַד שֶׁתִּרְאֶה דָּם. וְאֶצְלֵנוּ נָהוּג לְהַחְמִיר בָּזֶה. אֲבָל אִם פָּסְקוּ הַצִּירִים, תִּשְׁאַל רַב, כִּי אָז אֶפְשָׁר שֶׁיְּטַהֲרֶנָּה, מֵאַחַר שֶׁהוּכַח שֶׁלֹּא הָיְתָה יוֹלֶדֶת אָז. ↩

  3. רְאֵה שו"ע סי' קצד ס"א. ↩
