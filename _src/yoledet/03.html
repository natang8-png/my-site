וְהִנֵּה רמ״א בשו״ע (סי׳ קצד ס״א) מֵבִיא, שֶׁיֵּשׁ מְקוֹמוֹת שֶׁנָּהֲגוּ שֶׁאִשָּׁה יוֹלֶדֶת זָכָר טְמֵאָה אַרְבָּעִים יוֹם לְפָחוֹת וְיוֹלֶדֶת נְקֵבָה טְמֵאָה שְׁמוֹנִים יוֹם לְפָחוֹת, ואע״פ שֶׁלִּכְאוֹרָה זֶה הָפוּךְ מֵהַכָּתוּב בַּתּוֹרָה, שֶׁהֲרֵי בַּתּוֹרָה נִכְתַּב בְּפֵירוּשׁ שֶׁיֵּשׁ לַיּוֹלֶדֶת יְמֵי טוֹהַר, מ״מ, כָּךְ נָהֲגוּ בְּכַמָּה מְקוֹמוֹת וְכַמָּה הֶסְבֵּרִים נֶאֶמְרוּ בַּדָּבָר. אַחַד הַהֶסְבֵּרִים הוּא, דְּהִנֵּה הַתּוֹרָה טִיהֲרָה אֶת הַדָּמִים בִּימֵי טוֹהֳרָהּ, וְהַטַּעַם שֶׁטִּיהֲרָה הַתּוֹרָה – כֵּיוָן שֶׁזֶּהוּ זְמַן שֶׁצְּפוּיָה לְדַמֵּם בּוֹ מִצַּד הַלֵּידָה וְהַתּוֹרָה טִיהֲרַתּוּ מֵאַחַר שֶׁיֵּשׁ לָזֶה סִיבָּה מְיוּחֶדֶת וְשׁוֹנָה מִשְּׁאָר דַּם הָאִשָּׁה.

וּמֵאַחַר שֶׁהַיּוֹם נָהוּג לְטַמֵּא כָּל דָּם [כְּדֵי שֶׁלֹּא יָבוֹאוּ לְזַלְזֵל בְּדָם שֶׁמְּטַמֵּא], ע״כ חָשְׁשׁוּ וְאָסְרוּ אֶת הָאִשָּׁה בְּאוֹתָם יָמִים, מִפְּנֵי שֶׁכָּאָמוּר, הַתּוֹרָה הֵעִידָה שֶׁמָּצוּי אֶצְלָהּ דָּם וְחָשְׁשׁוּ שֶׁמָּא תִּרְאֶה דָּם וְלֹא תִּהְיֶה מוּדַעַת לְכָךְ, כֵּיוָן שֶׁבַּיָּמִים אֵלּוּ מָצוּי אֶצְלָהּ דָּם, כנ״ל.

אוּלָם מִנְהָג זֶה אֵינוֹ נָהוּג בִּקְהִילּוֹת הָאַשְׁכְּנַזִּים וְגַם לֹא בְּקֶרֶב אנ״ש אִם כִּי יֵשׁ קְצָת מִקְּהִילּוֹת הַסְּפָרַדִּים שֶׁנָּהֲגוּ כֵּן.

וְלָכֵן, אִם אִשָּׁה נָהֲגָה כֵּן אוֹ שֶׁאִימָּהּ נָהֲגָה כֵּן – תִּשְׁאַל רַב הַאִם עָלֶיהָ לְהַמְשִׁיךְ בַּמִּנְהָג אוֹ לְבַטְּלוֹ, וְאִם הִיא רוֹצָה שֶׁלֹּא לִנְהוֹג כֵּן, הַאִם צְרִיכָה [הִיא וְגַם בַּעְלָהּ] הַתָּרַת נְדָרִים.
