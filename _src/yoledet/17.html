ז. חוֹבָה לְהִשְׁתַּדֵּל לְהָבִיא יְלָדִים לָעוֹלָם כְּכָל שֶׁרַק נִיתָּן כְּמוֹ שֶׁנִּפְסַק בַּהֲלָכָה, וְהָרַבִּי הֶאֱרִיךְ לְהַפְלִיג בַּחֲשִׁיבוּת הַדָּבָר, שֶׁזֶּה מְקָרֵב אֶת הַגְּאוּלָּה, מוֹסִיף בִּשְׁלוֹם בַּיִת בַּמִּשְׁפָּחָה, מוֹסִיף בְּשֶׁפַע פַּרְנָסָה וכו', וְשֶׁכָּל אֶחָד יִשְׁתַּדֵּל שֶׁיִּהְיֶה לוֹ מִנְיָן מִמִּשְׁפַּחְתּוֹ, וְאִם יֵשׁ לוֹ מִנְיָן יִרְאֶה שֶׁיִּהְיוּ לוֹ י"ב שְׁבָטִים, וְאִם יֵשׁ לוֹ י"ב שְׁבָטִים יִהְיֶה מוֹסִיף וְהוֹלֵךְ בְּטוֹב.

וְאִם מִפְּנֵי סִיבָּה רוֹצִים לִמְנוֹעַ לִזְמַן מוּגְבָּל תּוֹסֶפֶת יְלָדִים - צָרִיךְ לִשְׁאוֹל רַב [וְאִם יֵשׁ חֲשָׁשׁ לְנֶזֶק בִּבְרִיאוּת הַגּוּף אוֹ הַנֶּפֶשׁ אִם יִהְיֶה הֵרָיוֹן נוֹסָף בְּקָרוֹב, חַיָּיבִים לִשְׁאוֹל רַב] וּלְנַמֵּק אֶת הַסִּיבָּה וּלְהִישָּׁמַע לִפְסַק הָרַב, וְכַמּוּבָן שֶׁלֹּא לְהוֹסִיף עַל הַזְּמַן שֶׁהָרַב הִתִּיר לְהִימָּנַע מֵהֵרָיוֹן, וְגַם לְבָרֵר אֵצֶל הָרַב [וְרוֹפֵא] בְּאֵיזֶה אוֹפֶן נִיתָּן וּמוּתָּר לִמְנוֹעַ [אֲבָל אֵין לַעֲשׂוֹת, ח"ו, דָּבָר שֶׁהוּא מְנִיעָה תְּמִידִית שֶׁאֵינָהּ נִיתֶּנֶת לְשִׁינּוּי].

ח. קַיָּים אִיסּוּר בָּרוּר לְהַפִּיל בְּכַוָּונָה אוֹ בְּרַשְׁלָנוּת עוּבָּרִים וְזֶה נִכְלָל בְּאִיסּוּר "לֹא תִּרְצָח". וְגַם אִם הָיָה הֶיתֵּר לִמְנוֹעַ הֵרָיוֹן וְקָרָה שֶׁהִיא נִתְעַבְּרָה בְּכָל זֹאת - קַיָּים אִיסּוּר בָּרוּר לְהַפִּיל. וְהַסְּבוּרָה שֶׁיֵּשׁ לָהּ הֶכְרֵחַ אוֹ אִם לְדַעַת הָרוֹפֵא יֵשׁ הֶכְרֵחַ מִפְּנֵי סַכָּנָה לָאֵם [אוֹ נֶזֶק מַמָּשִׁי בִּבְרִיאוּת הַגּוּף אוֹ הַנֶּפֶשׁ] - עָלֶיהָ לְהִתְיָיעֵץ עִם רַב מוּסְמָךְ וְגָדוֹל בַּתּוֹרָה. וִידוּעָה תְּפִילַּת כֹּהֵן גָּדוֹל ביוהכ"פ שֶׁלֹּא תַּפִּיל אִשָּׁה פְּרִי בִּטְנָהּ, וְאַדְּרַבָּה, שֶׁיִּתְבָּרְכוּ כָּל בֵּית יִשְׂרָאֵל בִּילָדִים בְּרִיאִים וְטוֹבִים ועי"ז נְקָרֵב אֶת הַגְּאוּלָּה הַשְּׁלֵימָה בְּקָרוֹב מַמָּשׁ.
