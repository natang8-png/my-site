וּלְהַלָּן דִּינֵי יוֹלֶדֶת וּמַפֶּלֶת בִּקְצָרָה:

א. יוֹלֶדֶת נֶאֱסֶרֶת מֵרֶגַע שֶׁיּוֹצֵאת מִמֶּנָּה טִפַּת דָּם, אַף אִם זֶה לִפְנֵי תַּהֲלִיךְ הַלֵּדָה אוֹ, חָלִילָה, בְּמַהֲלַךְ הַהֵרָיוֹן [וְכַמּוּבָן שֶׁיֵּשׁ לְהִתְיַעֵץ מִיָּד עִם רוֹפֵא בְּכָל מִקְרֶה שֶׁל דִּמּוּם בְּהֵרָיוֹן אוֹ בְּהַתְחָלַת הַלֵּדָה].

ב. הַיּוֹלֶדֶת נִטְמֵאת אִם הָיָה דִּמּוּם מִגּוּפָהּ. וְאִם הָיְתָה פְּתִיחַת הָרֶחֶם בִּגְלַל צִירֵי הַלֵּדָה, תִּשְׁאַל רַב. וְכֵן אִם אֵינָהּ יְכוֹלָה לַעֲמֹד עַל רַגְלֶיהָ בִּגְלַל הַצִּירִים, הִיא טְמֵאָה.

וּמוּבָן שֶׁאִם זְקוּקָה לְסִיּוּעַ, כְּגוֹן לַעֲלוֹת לָרֶכֶב, צָרִיךְ הַבַּעַל לַעֲזֹר לָהּ בְּמִדַּת הַהֶכְרֵחַ, כְּלוֹמַר, לְלֹא קִרְבָה פִיזִית שֶׁאֵינָהּ הֶכְרֵחִית. וְגַם הַסִּיּוּעַ הַהֶכְרֵחִי כְּדֵי לְהַגִּיעַ לְבֵית הָרְפוּאָה לָלֶדֶת צָרִיךְ שֶׁיִּהְיֶה ע"י בֶּגֶד, וּלְהִשְׁתַּדֵּל שֶׁלֹּא לִנְגֹּעַ יְשִׁירוֹת בִּבְשָׂרָהּ.

וְאִם נִפְסַק תַּהֲלִיךְ הַלֵּדָה וְלֹא רָאֲתָה דָּם, יֵשׁ לִשְׁאֹל רַב [וְאִם יָצְאָה מִמֶּנָּה כַּמּוּת גְּדוֹלָה שֶׁל מַיִם וְלֹא נִרְאָה דָּם, תִּשְׁאַל רַב].

ג. לְאוֹר הנ"ל, דַּעַת הָרַבִּי שֶׁאֵין לַבַּעַל לִהְיוֹת נוֹכֵחַ בַּחֲדַר הַלֵּדָה אוֹ בַּחֲדַר הַצִּירִים [כַּנִּרְאֶה שֶׁהָאִסּוּר הוּא מֵהַזְּמַן שֶׁהָאִשָּׁה פּוֹשֶׁטֶת בְּגָדֶיהָ הָרְגִילִים וּמִתְכּוֹנֶנֶת לָלֶדֶת, בִּמְקוֹמוֹת שֶׁמַּחֲלִיפִים אֶת הַבְּגָדִים זְמַן רַב יַחֲסִית לִפְנֵי הַלֵּדָה. אַךְ בִּמְקוֹמוֹת שֶׁנָּהוּג לִלְבֹּשׁ חָלוּק סָמוּךְ מַמָּשׁ לַלֵּדָה, יִתָּכֵן שֶׁהַבַּעַל צָרִיךְ לָלֶכֶת גַּם לִפְנֵי הַחְלָפַת הַבְּגָדִים].

ד. אַחַר הַלֵּדָה, עַל הָאִשָּׁה לְהַמְתִּין עַד שֶׁפּוֹסֵק הַדָּם, וְלִמְנוֹת שִׁבְעָה יָמִים נְקִיִּים וְלִטְבֹּל, כְּמוֹ שֶׁהוּבָא לְעֵיל.

וְיֵשׁ לְהַזְכִּיר שֶׁבְּלֵדַת בֵּן עָלֶיהָ לְהַמְתִּין חֲמִשָּׁה יָמִים מִתְּחִלַּת הָאִסּוּר וְלִמְנוֹת שִׁבְעָה יָמִים נְקִיִּים, וּבְלֵדַת בַּת עָלֶיהָ לְהַמְתִּין שִׁבְעָה יָמִים אַחַר הַלֵּדָה לִפְנֵי סְפִירַת שִׁבְעָה יָמִים נְקִיִּים [אַךְ בְּדֶרֶךְ כְּלָל לוֹקֵחַ זְמַן מְמֻשָּׁךְ עַד שֶׁפּוֹסֵק הַדָּם, וְרַק אָז פּוֹסֶקֶת בְּטָהֳרָה, סוֹפֶרֶת שִׁבְעָה יָמִים וְטוֹבֶלֶת].

ה. אִשָּׁה שֶׁהִפִּילָה, ח"ו, אִם זוֹ הָיְתָה נְקֵבָה, אוֹ אִם אֵין יָדוּעַ אִם הַנֵּפֶל בֵּן אוֹ בַּת, עָלֶיהָ לְהַמְתִּין שִׁבְעָה יָמִים אַחַר הַהַפָּלָה, וְרַק אח"כ לִסְפֹּר שִׁבְעָה יָמִים נְקִיִּים.

וְאִם יוֹדַעַת שֶׁהַנֵּפֶל הָיָה זָכָר, מַמְתִּינָה חֲמִשָּׁה יָמִים מִתְּחִלַּת טֻמְאָתָהּ וְאח"כ סוֹפֶרֶת שִׁבְעָה יָמִים נְקִיִּים.

וְאִם הַהַפָּלָה הָיְתָה בַּשָּׁבוּעוֹת הָרִאשׁוֹנִים, תִּשְׁאַל רַב הַאִם יְכוֹלָה לְהִסְתַּפֵּק בְּהַמְתָּנַת חֲמִשָּׁה יָמִים וְאח"כ שִׁבְעָה יָמִים נְקִיִּים.

ו. נָשִׁים רַבּוֹת עוֹשׂוֹת בְּדִיקַת רוֹפֵא אַחַר הַלֵּדָה, וְרָאוּי לַעֲשׂוֹת זֹאת לִפְנֵי הֶפְסֵק הַטָּהֳרָה. אַךְ אִם נִבְדְּקָה אַחַר הֶפְסֵק הַטָּהֳרָה, יֵשׁ לִשְׁאֹל אֶת הָרוֹפֵא לְאֵיזֶה מָקוֹם חָדַר בַּבְּדִיקָה וְלִשְׁאֹל רַב, וְגַם אִם יָצָא דָּם יֵשׁ לִשְׁאֹל רַב [וּלְבָרֵר הַאִם הוּא דָּם פְּנִימִי אוֹ חִיצוֹנִי].

וְאִם הָיָה לָהּ דִּמּוּם שֶׁיִּתָּכֵן שֶׁהוּא מֵהַתְּפָרִים וְכַדּוֹמֶה, תִּתְיַעֵץ עִם רַב [גַּם אִם זֶה קָרָה שֶׁלֹּא בִּבְדִיקַת רוֹפֵא].
