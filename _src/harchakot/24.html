לה. נָהֲגוּ הַנָּשִׁים שֶׁלֹּא לֵילֵךְ לְבֵית־הַכְּנֶסֶת בִּזְמַן הַדִּמּוּם [אֲבָל כְּשֶׁפָּסַק הַדִּמּוּם אע"פ שֶׁעֲדַיִן לֹא עָשְׂתָה הֶפְסֵק טָהֳרָה – מֻתֶּרֶת לֵילֵךְ לְבֵית־הַכְּנֶסֶת].

וְאִם נוֹהֶגֶת לָלֶכֶת בְּכָל שַׁבָּת לְבֵית־הַכְּנֶסֶת אוֹ שֶׁיֵּשׁ לָהּ סִבָּה חֲזָקָה לֵילֵךְ לְבֵית־הַכְּנֶסֶת, כְּגוֹן בְּיָמִים נוֹרָאִים, חַגִּים, אוֹ שִׂמְחָה הַנֶּעֱרֶכֶת בְּבֵית־הַכְּנֶסֶת שֶׁצְּרִיכָה לְהִשְׁתַּתֵּף בָּהּ – מַתִּירִים לָהּ לָלֶכֶת לְבֵית־הַכְּנֶסֶת. וּמ"מ, תִּזָּהֵרְנָה שֶׁלֹּא לִרְאוֹת סֵפֶר תּוֹרָה בִּזְמַן הַהַגְבָּהָה [כְּשֶׁפּוֹתְחִין הַכְּתָב].

וְכֵן מֻתֶּרֶת לְהִשְׁתַּתֵּף בְּשִׁעוּר תּוֹרָה, הַרְצָאָה אוֹ הִתְוַעֲדוּת לְנָשִׁים. וּבְוַדַּאי שֶׁמֻּתָּר לַמַּרְצֶה וכיו"ב לָבוֹא לְבֵית־הַכְּנֶסֶת לְהַרְצוֹת לְנָשִׁים גַּם אִם הַהַרְצָאָה נֶעֱרֶכֶת בְּבֵית־הַכְּנֶסֶת עַצְמוֹ וְוַדַּאי מְקִלִּים אִם נֶעֱרֶכֶת בְּעֶזְרַת נָשִׁים.

וְכֵן יֵשׁ לְהַתִּיר לְהַכְנִיס נְשֵׁי וּבְנוֹת יִשְׂרָאֵל לְבֵית־הַכְּנֶסֶת לְצֹרֶךְ הַרְצָאָה וכיו"ב כְּדֵי לְקָרְבָן לִשְׁמִירַת תּוֹרָה וּמִצְווֹת35.

  1. רְאֵה שוע"ר או"ח סי' פח ס"ב. וּבְקִצּוּר דִּינֵי טָהֳרָה, קה"ת, סי' ג סכ"ו, שֶׁכְּשֶׁפָּסַק זִלּוּף הַדָּם מֻתֶּרֶת לְהִכָּנֵס לביהכ"נ.

    וּבְסֵפֶר שַׁעֲרֵי הֲלָכָה וּמִנְהָג ח"א עמ' רב-רג מוּבָא מֵהָרַבִּי, שֶׁאֵין לִמְנֹעַ מִנָּשִׁים וּבָנוֹת לָבוֹא לְכִנּוּסִים בביהכ"ס [אַף בִּזְמַן הַדִּמּוּם, מִצַּד צֹרֶךְ הַמִּצְוָה שֶׁבָּזֶה].

    וּמִן הַדִּין מֻתֶּרֶת לִיגַּע בְּדִבְרֵי קְדֻשָּׁה, כְּגוֹן סְפָרִים, תְּפִלִּין וּמְזוּזוֹת גַּם בִּזְמַן הַדִּמּוּם, כְּמ"ש בשוע"ר שָׁם, וְיֵשׁ הַמַּחְמִירוֹת שֶׁלֹּא לִנְגֹּעַ בִּזְמַן הַדִּמּוּם בִּתְפִלִּין וּמְזוּזוֹת [בְּצוּרָה יְשִׁירָה בְּיָדָהּ בְּדִבְרֵי קְדֻשָּׁה, אֶלָּא נוֹגַעַת ע"י חֲצִיצָה כְּמוֹ בֶּגֶד, נַיְלוֹן וכד']. ↩
