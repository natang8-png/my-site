ז. לֹא יוֹשִׁיטוּ זֶה לָזֶה אֲפִלּוּ דָּבָר אָרֹךְ שֶׁאֵין חֲשָׁשׁ שֶׁל נְגִיעָה, וְכֵן לֹא יָרִימוּ דָּבָר בְּיַחַד [וּלְעִנְיַן דָּבָר כָּבֵד רְאֵה בַּהֶעָרָה], וְכֵן לֹא יִזְרְקוּ דָּבָר זֶה לָזֶה. לִזְרֹק לָאֲוִיר כְּלַפֵּי מַעְלָה וְהַשֵּׁנִי נוֹטְלוֹ כְּשֶׁהוּא נוֹפֵל, אע"פ שֶׁיֵּשׁ מָקוֹם לְהַתִּיר, מ"מ, הַמַּחְמִיר גַּם בָּזֶה תָּבוֹא עָלָיו בְּרָכָה. וְלֹא יִתְּנוּ אוֹ יִקְחוּ דָּבָר מִכִּיס אוֹ מִתִּיק אוֹ סַל שֶׁל הַשֵּׁנִי. וְיֵשׁ שֶׁאָסְרוּ לְהַנִּיחַ אוֹ לָקַחַת מֵעֲגָלַת קְנִיּוֹת כְּשֶׁהַשֵּׁנִי מַחֲזִיק בָּהּ.

וְכֵן אָסוּר לְהַדְלִיק סִיגַרְיָה מֵאֵשׁ שֶׁהָאִשָּׁה מַחֲזִיקָה אַךְ מֻתָּר לִרְאוֹת בְּנֵר הַבְדָּלָה שֶׁהָאִשָּׁה מַחֲזִיקָה.

וְכֵן רָאוּי שֶׁלֹּא לִקְרֹא מִסֵּפֶר אֶחָד [מֵחֲשַׁשׁ שֶׁיָּבוֹאוּ לִידֵי נְגִיעָה], אַךְ בִּשְׁעַת הַדְּחָק אֶפְשָׁר לִקְרֹא בְּאוֹתוֹ סֵפֶר תּוֹךְ זְהִירוּת שֶׁלֹּא יִתְקָרְבוּ זֶה לָזוֹ וְיָבוֹאוּ לִידֵי נְגִיעָה7.

ח. אֲסוּרִים לִמְסֹר תִּינוֹק קָטָן אוֹ תִּינוֹק יָשֵׁן מִיָּד לְיָד, אֲבָל תִּינוֹק גָּדוֹל וְעֵרָנִי וּמוֹסֵר עַצְמוֹ מִיָּד לְיָד יֵשׁ מַתִּירִים. וּבִשְׁעַת הַצֹּרֶךְ אֶפְשָׁר לְהָקֵל, כְּגוֹן תִּינוֹק גָּדוֹל בְּעֵרֶךְ מִגִּיל שָׁנָה וּמַעְלָה כְּשֶׁהוּא עֵר וּמוֹסֵר עַצְמוֹ מִיָּד לְיָד וְאֵינוֹ נִרְגָּע עַד שֶׁיַּעֲבִירוּהוּ הוֹרָיו מִיָּד לְיָד.

וְכֵן אָסוּר לְהִשְׁתַּעֲשֵׁעַ עִם הַיֶּלֶד שֶׁנִּמְצָא אֵצֶל בֶּן אוֹ בַּת הַזּוּג. וְכֵן אֵין לְהַאֲכִילוֹ אוֹ לָתֵת לוֹ דָּבָר כְּשֶׁהוּא בְּיַד בֶּן אוֹ בַּת הַזּוּג, אַךְ בִּשְׁעַת הַדְּחָק כְּשֶׁאֵין הָאֶחָד מַצְלִיחַ לְהַאֲכִילוֹ אוֹ לָתֵת לוֹ רְפוּאָה וכיו"ב – מֻתָּר לְהַאֲכִיל אֶת הַתִּינוֹק כְּשֶׁהוּא אֵצֶל בֶּן הַזּוּג, כַּמּוּבָן בִּזְהִירוּת, שֶׁלֹּא יָבוֹאוּ לִנְגִיעָה.

וְכֵן מֻתָּר לְהַנִּיחַ אֶת הַתִּינוֹק [בִּשְׁעַת הַצֹּרֶךְ] עַל מִטַּת הָאִשָּׁה אע"פ שֶׁהָאִשָּׁה נִמְצֵאת בְּמִטָּתָהּ, אַךְ לֹא עַל הַשְּׂמִיכָה שֶׁהִיא מִתְכַּסֵּית בָּהּ, וּכְשֶׁאֵין דֶּרֶךְ אַחֶרֶת מֻתָּר לוֹ לְהַנִּיחַ אֶת הַתִּינוֹק גַּם עַל הַשְּׂמִיכָה שֶׁהִיא מִתְכַּסֵּית בָּהּ [וּכְשֶׁאֵין כָּל בְּרֵירָה יָכוֹל לְהַנִּיחַ עַל בִּרְכֶּיהָ] אַךְ לֹא בִּמְסִירָה מִיָּד לְיָד וּלְהִזָּהֵר מִנְּגִיעָה8.
