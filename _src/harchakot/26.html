לז. בְּיוֹם הַכִּפּוּרִים וּבְיוֹם תִּשְׁעָה בְּאָב, גַּם אִם הָאִשָּׁה טְהוֹרָה, נָכוֹן לְהַחְמִיר לִנְהֹג בְּכָל הַהַרְחָקוֹת הַנַּ"ל [וְתַשְׁמִישׁ אָסוּר מֵעִקַּר הַדִּין].

[וּבִזְמַן אֲנִינוּת אוֹ אֲבֵלוּת, רח"ל, שֶׁל הָאִשָּׁה אוֹ שֶׁל הַבַּעַל, אֲסוּרִים בְּתַשְׁמִישׁ וּבְקֵרוּב בָּשָׂר וְכֵן לֹא לִישֹׁן בְּמִטָּה אַחַת]37.

  1. רְאֵה שוע"ר יו"ד סי' קפד סק"ו, וּבְאו"ח סי' תרטו ס"א, שֶׁיֵּשׁ לְהַחְמִיר בְּיוהכ"פ בְּכָל הַהַרְחָקוֹת בַּלַּיְלָה וּבַיּוֹם.

    וְשָׁם בְּסִי' מ קו"א סק"ב, שֶׁגַּם גַּבֵּי ט' בְּאָב אֵין לְחַלֵּק בֵּין לַיְלָה לְיוֹם [וְזוֹ לְשׁוֹן רַבֵּנוּ שָׁם בְּס"ח בַּמֻּסְגָּר: "שֶׁהֲרֵי מִן הַדִּין מֻתָּר לִישֹׁן עִם אִשְׁתּוֹ בת"ב וְאֵין חוֹשְׁשִׁין שֶׁמָּא יִשְׁכַּח (וִישַׁמֵּשׁ), אֶלָּא שֶׁנָּכוֹן הַדָּבָר לְהַחְמִיר"].

    וְעַל כֵּן, אִם יֵשׁ בְּרִית מִילָה בת"ב אוֹ בְּיוהכ"פ, יֵשׁ לְהִזָּהֵר שֶׁה"קוואטער" [הַזּוּג שֶׁמַּעֲבִירִים אֶת הַיֶּלֶד הַנִּמּוֹל מֵאִמּוֹ לְאָבִיו כְּדֵי לְמוּלוֹ] לֹא יִמְסְרוּ מִיָּד לְיָד אֶלָּא יַנִּיחוּהוּ בֵּינֵיהֶם עַל אֵיזֶה מָקוֹם, וְאָז הַשֵּׁנִי יִטְּלֶנּוּ (רְאֵה הֶעָרָה 8).

    וְעַל כֵּן אֶפְשָׁר לָקַחַת אָז גַּם זוּג בִּזְמַן שֶׁהָאִשָּׁה טְמֵאָה, כֵּיוָן שֶׁכַּנַּ"ל, בְּכָל מַצָּב אֵין מַעֲבִירִים מִיָּד לְיָד בת"ב וּבְיוהכ"פ. אוֹ שֶׁיִּקְחוּ אָב וּבִתּוֹ אוֹ אָח וַאֲחוֹתוֹ שֶׁמֻּתָּרִים לִמְסֹר מִיָּד לְיָד גַּם בְּיָמִים אֵלּוּ. ↑ לטקסט
