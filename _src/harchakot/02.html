בַּתַּלְמוּד (שבת יג, א) מוּבָא מַעֲשֶׂה בְּתַלְמִיד־חָכָם אֶחָד שֶׁנִּפְטַר בְּגִיל צָעִיר, ל"ע, וְהָיְתָה אִשְׁתּוֹ נוֹטֶלֶת אֶת הַתְּפִלִּין שֶׁלּוֹ וְסוֹבֶבֶת בְּבָתֵּי־כְּנֵסִיּוֹת וּבְבָתֵּי־מִדְרָשׁוֹת וְטוֹעֶנֶת, שֶׁהֲרֵי נֶאֱמַר עַל הַתּוֹרָה "כִּי הוּא חַיֶּיךָ וְאֹרֶךְ יָמֶיךָ" (דברים ל, כ), בַּעְלָהּ שֶׁלָּמַד הַרְבֵּה תּוֹרָה, הֵיכָן אוֹתָהּ אֲרִיכוּת יָמִים הַמֻּבְטַחַת לוֹ? וְלֹא הָיָה אָדָם שֶׁהֱשִׁיבָהּ תְּשׁוּבָה עַל טַעֲנָה זוֹ, עַד שֶׁהִתְגַּלָּה אֵלֶיהָ אֵלִיָּהוּ הַנָּבִיא וְאַחַר שֶׁהֵשִׂיחָה בְּפָנָיו כָּל הַמְּאֹרָע, שָׁאַל אוֹתָהּ: "בִּתִּי, כֵּיצַד נָהַג עִמָּךְ בִּזְמַן הַנִּדָּה?" עָנְתָה הָאִשָּׁה, שֶׁנָּהַג עִמָּהּ כָּל הַהַרְחָקוֹת הַכְּתוּבוֹת בַּהֲלָכָה, כְּגוֹן שֶׁלֹּא אָכַל עִמָּהּ בְּאוֹתוֹ שֻׁלְחָן [אִם לֹא הָיָה דָּבָר שֶׁל הֶכֵּר בֵּינֵיהֶם] וְכוּ'.

שׁוּב שְׁאָלָהּ אֵלִיָּהוּ הַנָּבִיא: "בִּימֵי לִבּוּנֵךְ כֵּיצַד נָהַג?" כְּלוֹמַר, כֵּיצַד הִתְנַהֵג עִמָּךְ בִּזְמַן שִׁבְעָה נְקִיִּים [שֶׁאָז הָאִשָּׁה לוֹבֶשֶׁת לְבָנִים]?

עָנְתָה, שֶׁבְּאוֹתָם יָמִים לֹא הִקְפִּיד עַל הַהַרְחָקוֹת בַּסְּעוּדָה וְכוּ' אֲבָל וַדַּאי נִזְהַר מִתַּשְׁמִישׁ כו'.

עָנָה לָהּ אֵלִיָּהוּ הַנָּבִיא: "בָּרוּךְ הַמָּקוֹם שֶׁהֲרָגוֹ, שֶׁהֲרֵי נֶאֱמַר 'וְאֶל אִשָּׁה בְּנִדַּת טֻמְאָתָהּ לֹא תִקְרַב' גו'". כְּלוֹמַר, אֵלִיָּהוּ הַנָּבִיא הִצְדִּיק אֶת הָעֹנֶשׁ הֶחָמוּר שֶׁלּוֹ בְּזֶה שֶׁטָּעָה בַּהֲלָכָה וְלֹא נִזְהַר בַּהַרְחָקוֹת בְּשִׁבְעָה נְקִיִּים.

מִזֶּה רוֹאִים עַד הֵיכָן הַדְּבָרִים מַגִּיעִים! אֵלִיָּהוּ אַף הִשְׁתַּמֵּשׁ בַּבִּטּוּי "בָּרוּךְ הַמָּקוֹם שֶׁהֲרָגוֹ!", כְּלוֹמַר, טוֹבָה נַעֲשֵׂית לוֹ שֶׁנֶּעֱנַשׁ בעוה"ז בְּמִיתָה בְּגִיל צָעִיר כְּדֵי שֶׁיִּזְכֶּה לָעוֹלָם הַבָּא.
