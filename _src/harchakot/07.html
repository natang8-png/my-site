ו. לֹא יִסְתַּכֵּל בָּהּ בִּמְקוֹמוֹת שֶׁדַּרְכָּם לִהְיוֹת מְכֻסִּים, כְּגוֹן שַׂעֲרוֹת רֹאשָׁהּ. אֲבָל בִּמְקוֹמוֹת שֶׁדַּרְכָּם לִהְיוֹת גְּלוּיִים, כְּגוֹן פָּנֶיהָ – מֻתָּר לַבַּעַל לִרְאוֹת. אַךְ רָאוּי וְנָכוֹן שֶׁלֹּא לְהִתְבּוֹנֵן בְּמַרְאֵה פָּנֶיהָ בִּזְמַן הָאִסּוּר וְכֵן אָסוּר לִשְׁמֹעַ קוֹל זֶמֶר שֶׁלָּהּ6.

  1. שו"ע סי' קצה ס"ז וּבְסֵפֶר פִּסְקֵי דִּינִים שָׁם ס"ז מֵבִיא שֶׁנָּכוֹן שֶׁלֹּא לְהִסְתַּכֵּל גַּם בִּמְקוֹמוֹת הַגְּלוּיִים. וּמַשְׁמָע שֶׁהַכַּוָּנָה שֶׁלֹּא לְהִתְבּוֹנֵן בְּפָנֶיהָ, אֲבָל רְאִיָּה לְבַד – מֻתָּר, עכ"פ בִּמְקוֹמוֹת הַגְּלוּיִים וּרְאֵה עד"ז בשוע"ר או"ח סי' קכח סל"ו.

    וּלְגַבֵּי שְׂעַר רֹאשָׁהּ דָּנִים הַמְחַבְּרִים בְּאִשָּׁה שֶׁרְגִילָה לֵילֵךְ בַּבַּיִת בְּגִלּוּי רֹאשׁ, הַאִם זֶה בִּכְלַל מְקוֹמוֹת הַמְכֻסִּים. וּמִכֵּיוָן שֶׁאֶצְלֵנוּ מַקְפִּידִים לְכַסּוֹת רֹאשָׁהּ גַּם בַּבַּיִת, ע"כ וַדַּאי שֶׁיֵּשׁ אִסּוּר לְהִסְתַּכֵּל בְּשַׂעֲרוֹתֶיהָ.

    וּלְעִנְיַן קוֹל זֶמֶר שֶׁלָּהּ – כָּךְ כָּתְבוּ הָאַחֲרוֹנִים וְהוּבָא בְּקִצּוּר דִּינֵי טָהֳרָה, הוֹצָאַת קה"ת, פ"ג ה"ג. וְיֵשׁ שֶׁהֵבִיאוּ עפ"ז לְהַקְפִּיד שֶׁלֹּא תֹּאמַר בִּרְכַּת הַמָּזוֹן בְּקוֹל זֶמֶר כְּשֶׁבַּעְלָהּ שׁוֹמֵעַ, וּבְמַצָּב שֶׁאֵין לְבַעְלָהּ מָקוֹם אַחֵר שֶׁלֹּא יִשְׁמָעֶנָּה וְאֵינוֹ מִתְכַּוֵּן לֵהָנוֹת מִזִּמְרָתָהּ – אֵין אִסּוּר. וּכְמ"ש בָּזֶה בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סכ"ד-כה. ↩
