כט. הַבַּעַל וְאִשְׁתּוֹ אֲסוּרִים לְהַצִּיעַ הַמִּטָּה זל"ז בִּזְמַן טֻמְאָתָהּ. אִסּוּר זֶה כּוֹלֵל סִדּוּר הַשְּׂמִיכָה, הַכָּרִים וְסִדּוּר הַסָּדִין [אֲבָל מֻתָּר לְהָבִיא מִזְרָן לְמִטַּת בֶּן הַזּוּג]. וְכֵן מֻתָּרִים לְהַחֲלִיף צִפִּיּוֹת הַכָּרִים וְהַשְּׂמִיכוֹת אֶחָד בִּפְנֵי הַשֵּׁנִי. הָאִסּוּר הוּא רַק הֲכָנַת הַמִּטָּה לְשֵׁנָה אֲבָל אֵין אִסּוּר לְסַדֵּר אֶת הַמִּטָּה לֹא כַּהֲכָנָה לְשֵׁנָה אֶלָּא בְּאֹפֶן שֶׁהַחֶדֶר יֵרָאֶה מְסֻדָּר, אֲבָל לְסַדֵּר לִקְרַאת שֵׁנָה – אָסוּר אֲפִלּוּ אִם מְסַדְּרִים בַּבֹּקֶר עֲבוּר הַלַּיְלָה.

הָאִסּוּר הוּא רַק בִּפְנֵי בֶּן הַזּוּג. וְנֶחְלְקוּ הַפּוֹסְקִים הַאִם הַהֶתֵּר הוּא רַק אִם בֶּן הַזּוּג אֵינוֹ בַּחֶדֶר אוֹ שֶׁמֻּתָּר גַּם אִם הוּא בַּחֶדֶר וּמְסוֹבֵב פָּנָיו שֶׁלֹּא יִרְאֶה [וְאִם הִצִּיעוּ בְּאִסּוּר, הֲרֵי בְּדִיעֲבַד מֻתָּר לִישֹׁן ע"ז (וּרְאֵה בַּהֶעָרָה)]29.

  1. רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה סי"א. וּמִסִּגְנוֹן דְּבָרָיו מַשְׁמָע קְצָת, שֶׁאִם הַבַּעַל נִמְצָא – אָסוּר אע"פ שֶׁאֵינוֹ רוֹאֶה, שֶׁהֲרֵי מָצָא הֶתֵּר רַק שֶׁלֹּא בְּפָנָיו, וְלֹא כָּתַב שֶׁיְּסוֹבֵב פָּנָיו וְלֹא יִסְתַּכֵּל. וְאוּלַי בִּשְׁעַת הַדְּחָק נִתָּן לְהַתִּיר כְּשֶׁאֵין הַשֵּׁנִי רוֹאֶה וּכְמ"ש בְּסֵפֶר טָהֳרָה בַּהֲלָכָה פי"ד סק"ב, שֶׁרֹב הַפּוֹסְקִים מַתִּירִים אִם הַשֵּׁנִי אֵינוֹ רוֹאֶה.

    וּגְבֵי הַנָּחַת מִזְרָן עַל מִטַּת בֶּן הַזּוּג – אֶפְשָׁר שֶׁבְּמִזְרָן קַל מְאֹד יֵשׁ לֶאֱסֹר.

    בְּדִיעֲבַד, אִם הִצִּיעוּ בְּאִסּוּר – מֻתָּר לִישֹׁן עַל זֶה, כְּמ"ש בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סק"ח.

    וְיֵשׁ שֶׁהֶחְמִירוּ לִסְתֹּר אֶת סִדּוּר הַמִּטָּה שֶׁנַּעֲשָׂה בְּאִסּוּר, וֶהֱבִיאָם בְּסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פכ"ט סט"ו וּלְדַעְתּוֹ נָכוֹן לְהַחְמִיר בָּזֶה. ↑ לטקסט
