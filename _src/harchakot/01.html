בִּזְמַן שֶׁהָאִשָּׁה בְּטֻמְאַת נִדָּתָהּ נֶאֱסַר עַל בְּנֵי הַזּוּג לְהִתְקָרֵב זֶה לָזוֹ וְזוֹ לָזֶה, וְאִסּוּרִים אֵלּוּ נֶחְלָקִים לִשְׁנֵי סוּגִים כְּלָלִיִּים:

א. אִסּוּר תַּשְׁמִישׁ, שֶׁהוּא אִסּוּר תּוֹרָה, שֶׁנֶּאֱמַר "וְאֶל אִשָּׁה בְּנִדַּת טֻמְאָתָהּ לֹא תִקְרַב לְגַלּוֹת עֶרְוָתָהּ" (ויקרא יח, יט), וְעָנְשׁוֹ כָּרֵת לָעוֹבְרִים בְּמֵזִיד.

עֹנֶשׁ הַכָּרֵת כּוֹלֵל ג' דְּבָרִים חֲמוּרִים מְאֹד:

  1. א) מָוֶת בְּקִצּוּר שָׁנִים [עַד גִּיל 50];
  2. ב) מָוֶת לְלֹא יְלָדִים [שֶׁלֹּא יִוָּלְדוּ לוֹ יְלָדִים וְאִם כְּבָר נוֹלְדוּ – יָמוּתוּ בְּחַיָּיו, ר"ל];
  3. ג) הַנֶּפֶשׁ נִכְרֶתֶת מֵחַיֵּי הָעוֹלָם הַבָּא [מִגַּן עֵדֶן].

הַבֵּית יוֹסֵף (סי' קצה) מֵבִיא, כִּי אִסּוּר נִדָּה נִכְלָל בְּאִסּוּר גִּלּוּי עֲרָיוֹת שֶׁעֲלֵיהֶם נֶאֱמַר "יֵהָרֵג וְאַל יַעֲבֹר", כְּלוֹמַר, אִם כּוֹפִים גֶּבֶר לָבוֹא עַל [אִשְׁתּוֹ] נִדָּה וְאִם לֹא יַעֲשֶׂה כֵּן יַהַרְגוּהוּ – עָלָיו לִמְסֹר נַפְשׁוֹ לְמִיתָה וְלֹא לַעֲבֹר [אִשָּׁה אֵינָהּ צְרִיכָה לִמְסֹר עַצְמָהּ לְמִיתָה בָּזֶה, כִּי הִיא אֵינָהּ עוֹשָׂה שׁוּם פְּעֻלָּה מִצִּדָּהּ].

וְכָתַב אַדְמוּ"ר הַזָּקֵן (תניא, אגרת התשובה פ"ו), שֶׁבִּזְמַן הַגָּלוּת אֶפְשָׁר שֶׁהַחוֹטֵא יַאֲרִיךְ יָמִים, כִּי אע"פ שֶׁחַיָּיו נִכְרָתִים מִשָּׁרְשָׁם בַּקְּדֻשָּׁה, מ"מ יָכוֹל לְקַבֵּל חַיּוּת מִצַּד הַקְּלִפָּה. אוּלָם וַדַּאי שֶׁבְּיַד הָאָדָם לָשׁוּב בִּתְשׁוּבָה שְׁלֵמָה מֵעֹמֶק הַלֵּב, לְסַלֵּק הַשְׁפָּעַת הַקְּלִפָּה מֵעָלָיו וְלַחֲזֹר וּלְקַבֵּל חַיּוּת מֵהַקְּדֻשָּׁה, וּלְבַטֵּל מֵעָלָיו אֶת הָעֳנָשִׁים הַנַּ"ל.

וְדַע, שֶׁבְּנוֹסָף לְעֹנֶשׁ הָאִישׁ וְהָאִשָּׁה, חֵטְא זֶה גַּם פּוֹגֵם בַּיְלָדִים שֶׁנּוֹלְדוּ שֶׁלֹּא בְּטָהֳרָה וְגוֹרֵם לָהֶם טֶבַע רַע וּמְגֻשָּׁם [וְגַם עָלוּל לִגְרֹם לְחִסָּרוֹן בִּבְרִיאוּת הַגּוּף וְהַנֶּפֶשׁ, ח"ו (מסכת כלה רבתי פ"ב מ"ט. ראה רמב"ם הל' איסורי ביאה פט"ו ה"א)].

הָרַבִּי (ראה ליקוטי שיחות ח"ב עמ' 579. וראה גם באגרות קודש חי"ח עמ' תנב) גַּם מוֹצֵא בְּכָךְ אֶת אֶחָד הַהֶסְבֵּרִים לָעֻבְדָּה, שֶׁבַּדּוֹרוֹת הָאַחֲרוֹנִים יֶשְׁנָם בְּעַם יִשְׂרָאֵל אֲנָשִׁים שֶׁחוֹטְאִים בִּדְבָרִים שֶׁבֶּעָבָר יְהוּדִים לֹא הָיוּ נִכְשָׁלִים בָּהֶם, שֶׁזֶּה נִגְרָם מִפְּנֵי שֶׁהוֹרֵיהֶם לֹא שָׁמְרוּ עַל טָהֳרַת הַמִּשְׁפָּחָה וְכָךְ נוֹלְדוּ לָהֶם יְלָדִים גַּסִּים וּמְגֻשָּׁמִים, וְהַדְּבָרִים מַרְאִים עַד כַּמָּה חָשׁוּב שֶׁיְּצִירַת הָעֻבָּר תִּהְיֶה בְּטָהֳרָה.

אָמְנָם נִתָּן לַעֲשׂוֹת "תִּקּוּן" מְסֻיָּם גַּם כְּלַפֵּי הַיֶּלֶד, בְּכָךְ שֶׁהַהוֹרִים יַקְפִּידוּ מִכָּאן וְאֵילָךְ עַל הִלְכוֹת טָהֳרַת הַמִּשְׁפָּחָה וְאַף יָפִיצוּ אֶת הַמִּצְוָה הַזֹּאת.

בְּנוֹסָף לְעִנְיַן הַטָּהֳרָה חָשׁוּב מְאֹד גַּם עִנְיַן הַקְּדֻשָּׁה דְּהַיְנוּ שֶׁיְּצִירַת הָעֻבָּר תִּהְיֶה מִתּוֹךְ מַחֲשָׁבוֹת טוֹבוֹת וּקְדוֹשׁוֹת שֶׁל הוֹרָיו, כַּמּוּבָא בְּסֵפֶר הַתַּנְיָא (לקוטי אמרים פ"ב) בְּשֵׁם הַזֹּהַר (ח"ב קיב, א. ח"ג מט, ב. ובכ"מ), שֶׁהָעִקָּר תָּלוּי בָּזֶה שֶׁיְּקַדְּשׁוּ הַהוֹרִים עַצְמָם בִּזְמַן הַתַּשְׁמִישׁ שֶׁעי"ז מַמְשִׁיכִים לְבוּשׁ קָדוֹשׁ [כְּלוֹמַר, תְּכוּנוֹת שֶׁל קְדֻשָּׁה] לְנִשְׁמוֹת הַבָּנִים כו'.

ב. אִסּוּר נוֹסָף מִן הַתּוֹרָה הוּא קֵרוּב לָאִשָּׁה [בְּדֶרֶךְ חִבָּה], כְּמוֹ חִבּוּק וְנִשּׁוּק. גַּם אִסּוּר זֶה נִלְמָד מִדִּיּוּק לְשׁוֹן הַפָּסוּק "וְאֶל אִשָּׁה בְּנִדַּת טֻמְאָתָהּ לֹא תִקְרַב לְגַלּוֹת עֶרְוָתָהּ" (ויקרא יח, יט), כְּלוֹמַר, גַּם הַקִּרְבָה הַגּוּפָנִית נֶאֶסְרָה מֵאַחַר שֶׁעֲלוּלָה לְהָבִיא לִידֵי גִּלּוּי עֶרְוָה מַמָּשׁ [תַּשְׁמִישׁ].

חֲזַ"ל הוֹסִיפוּ הַרְחָקוֹת רַבּוֹת כְּדֵי שֶׁלֹּא יָבוֹאוּ לִידֵי קֵרוּב וְתַשְׁמִישׁ וְגַם הֵם חֲמוּרִים מְאֹד וּמִי שֶׁאֵינוֹ נִזְהָר בָּזֶה עָלוּל לְהִכָּשֵׁל בְּתַשְׁמִישׁ, שֶׁהוּא אִסּוּר כָּרֵת, כַּנַּ"ל.
