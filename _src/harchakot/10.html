י. אֲסוּרִים לֶאֱכֹל עַל שֻׁלְחָן אֶחָד לְלֹא הֶכֵּר בֵּינֵיהֶם, כִּדְלְקַמָּן. וְגַם אִם אוֹכְלִים אֲחֵרִים עִמָּהֶם אָסוּר. אֲבָל אִם מַפְרִיד אָדָם בֵּינֵיהֶם, כְּגוֹן שֶׁיּוֹשֵׁב הַבַּעַל וּלְיָדוֹ אָדָם נוֹסָף וְאַחַר כָּךְ הָאִשָּׁה - מֻתָּר אַף לְלֹא הֶכֵּר. וְכֵן אִם הֵם יוֹשְׁבִים עַל שֻׁלְחָן מְרֻחָקִים זֶה מִזֶּה, כָּךְ שֶׁאֵין אֶפְשָׁרוּת לְהוֹשִׁיט יָד מֵאֶחָד לַשֵּׁנִי - מֻתָּר10.

יא. הֶכֵּר יָכוֹל לִהְיוֹת כָּל דָּבָר שֶׁמַּנִּיחִים בֵּינֵיהֶם וְאֵין דַּרְכּוֹ לַעֲמֹד שָׁם עַל הַשֻּׁלְחָן, בִּגְלַל שֶׁאֵין אוֹכְלִים מִמֶּנּוּ אוֹ כְּלִי שֶׁאֵין זֶה מְקוֹמוֹ, וּבִתְנַאי שֶׁהוּא נִכָּר בְּמַבָּט רִאשׁוֹן. וּבִכְלִי הֲרֵי גַּם אִם מְקוֹמוֹ לְמָשָׁל בְּמֶרְכַּז הַשֻּׁלְחָן וּבִזְמַן הָאִסּוּר מַנִּיחִים אוֹתוֹ בֵּינֵיהֶם ה"ז הֶכֵּר, אֲבָל מַאֲכָל אוֹ מַשְׁקֶה וְכֵן כְּלִי שֶׁיֵּשׁ בּוֹ מַאֲכָל אוֹ מַשְׁקֶה שֶׁאוֹכְלִים מִמֶּנּוּ בַּסְּעוּדָה לֹא יוֹעִיל מַה שֶּׁיְּשַׁנּוּ אֶת מְקוֹמוֹ עַל הַשֻּׁלְחָן11.

יב. וְכֵן אִם רְגִילִים לֶאֱכֹל עַל מַפָּה אַחַת וּבִזְמַן הַטֻּמְאָה אוֹכְלִים עַל שְׁתֵּי מַפּוֹת נִפְרָדוֹת - הֲרֵי זֶה הֶכֵּר. אוֹ שֶׁאֶחָד יַנִּיחַ מַפָּה נוֹסֶפֶת לְהֶכֵּר אוֹ יְקַפֵּל אֶת הַמַּפָּה פַּעַם נוֹסֶפֶת, שֶׁזֶּה נֶחְשָׁב כְּמַפָּה נוֹסֶפֶת. אוֹ שֶׁיָּשִׂימוּ מַפִּית נְיָר אוֹ צַלַּחַת רְחָבָה מִתַּחַת לַצַּלַּחַת אִם אֵין רְגִילִים בָּזֶה, וְהַמַּפִּית אוֹ הַצַּלַּחַת הַנּוֹסֶפֶת נִכָּרִים בְּמַבָּט רִאשׁוֹן שֶׁבּוֹלְטִים מֵעֵבֶר לַצַּלַּחַת12.

  1. רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה ס"ג. וְדַעַת רֹב הַפּוֹסְקִים לְהַתִּיר אִם אָדָם יוֹשֵׁב בֵּינֵיהֶם וְכֵן אִם הֵם מְרֻחָקִים שֶׁא"א לָאֶחָד לְהַגִּיעַ בְּיָדוֹ לְצַלַּחְתּוֹ שֶׁל הַשֵּׁנִי מֻתָּר.

    אֲבָל בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סס"א מַחְמִיר שֶׁלֹּא לְהַתִּיר ע"י אָדָם שֶׁמַּפְסִיק בֵּינֵיהֶם וְכֵן מֵבִיא לְהַחְמִיר גַּם אִם הֵם מְרֻחָקִים כָּךְ שֶׁאֵין הָאֶחָד יָכוֹל לְהוֹשִׁיט יָדוֹ לְצַלַּחַת הַשֵּׁנִי. וְרַק בְּהִצְטָרֵף שְׁנֵיהֶם יַחַד, מַתִּיר בִּשְׁעַת הַדְּחָק, עַיֵּן שָׁם וּבַהַנִּסְמָן בַּהֶעָרוֹת. ↑ לטקסט

  2. . רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה ס"ג. וְעד"ז בָּרמ"א יו"ד סו"ס פח. וְלִכְאוֹרָה מַשְׁמָע, שֶׁאִם יֶשְׁנוֹ סוּג אוֹ צֶבַע כֵּלִים מְיֻחָד לִימֵי הַטֻּמְאָה - אי"ז מוֹעִיל. וְכֵן לֹא רָאִיתִי שֶׁהֻזְכַּר הֶתֵּר זֶה בַּפּוֹסְקִים. וּלְעִנְיַן שִׁנּוּי מִקּוּם הַכְּלִי רְאֵה גַּם בִּפְרִי מְגָדִים יו"ד סי' פח שִׂפְתֵי דַּעַת סק"ז. ↑ לטקסט

  3. רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה ס"ג. וּבְסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סְעִיפִים נח-נט. וְכָתְבוּ הַמְחַבְּרִים, שֶׁגַּם אִם הַבַּעַל וְהָאִשָּׁה אוֹכְלִים זֶה חָלָב וְזוֹ בָּשָׂר, כָּךְ שֶׁחַיָּבִים בְּהֶכֵּר מִצַּד הִלְכוֹת בָּשָׂר בְּחָלָב כַּמְבֹאָר בשו"ע יו"ד סי' פח - הֶכֵּר זֶה מוֹעִיל גַּם לְהֶפְסֵק בִּזְמַן טֻמְאָה, כֵּיוָן שֶׁבְּפֹעַל יֵשׁ מַה שֶּׁמַּפְסִיק וּמַפְרִיד בֵּין הַבַּעַל וְהָאִשָּׁה. ↑ לטקסט
