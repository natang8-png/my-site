כח. מֻתָּר לִתֵּן מַתָּנָה לָאִשָּׁה בִּזְמַן טֻמְאָתָהּ, אַךְ מַתָּנָה חֲשׁוּבָה רָאוּי לָתֵת רַק בִּימֵי טָהֳרָתָהּ אוֹ לָתֵת בִּימֵי טֻמְאָה בְּהַשְׁאָלָה וּלְהַתְנוֹת שֶׁתִּזְכֶּה בַּמַּתָּנָה אַחַר טָהֳרָתָהּ. אֲבָל וַדַּאי מֻתָּר לִקְנוֹת פְּרָחִים לַבַּיִת לִכְבוֹד שַׁבָּת וְיו"ט אִם רָגִיל בָּזֶה28.

  1. בשו"ע לֹא הֻזְכַּר אִסּוּר עַל נְתִינַת מַתָּנָה וּלְדַעַת רֹב הַפּוֹסְקִים מֻתָּר לִתֵּן מַתָּנָה בִּימֵי טֻמְאָתָהּ אַךְ בְּמַתָּנָה חֲשׁוּבָה יֵשׁ לְהַחְמִיר.

    וּרְאֵה בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד ס"ק, שֶׁמ"מ יֵשׁ לְהַתִּיר בְּעֵת הַצֹּרֶךְ אִם מְנִיעַת הַמַּתָּנָה עֲלוּלָה לְהָבִיא לִידֵי קְפֵידָא [וּבִפְרָט לְאַחַר לֵדָה יֵשׁ לִתֵּן מַתָּנָה]. וְיֵשׁ שֶׁיָּעֲצוּ לָתֵת לָהּ בֵּנְתַיִם בְּתוֹר הַשְׁאָלָה וְתוּכַל לְהִשְׁתַּמֵּשׁ בַּמַּתָּנָה מִיָּד, אַךְ תִּזְכֶּה בָּהּ בְּפֹעַל רַק אַחַר טָהֳרָתָהּ. וּבְסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פל"ב ס"ה מְחַלֵּק, שֶׁרַק בְּמַתָּנָה חֲשׁוּבָה אוֹ בְּחֵפֶץ שֶׁל חִבָּה יֵשׁ לְהַחְמִיר.

    וְשָׁם בְּס"ו מַתִּיר לִשְׁלֹחַ פְּרָחִים אִם רָגִיל בָּזֶה בְּכָל שַׁבָּת וְיו"ט, אֲבָל אִם אֵינוֹ רָגִיל בָּזֶה - יֵשׁ לְהַחְמִיר. ↑ לטקסט
