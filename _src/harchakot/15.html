כב. אֲסוּרִים לֵישֵׁב עַל סַפְסָל אוֹ מִטָּה מִתְנַדְנֶדֶת, כְּלוֹמַר שֶׁאֶחָד מַרְגִּישׁ בְּכֹבֶד הַשֵּׁנִי, שֶׁמִּפְּנֵי כָּבְדּוֹ הַמִּטָּה שׁוֹקַעַת קְצָת - בֵּין אִם זֶה מִזְרָן אֶחָד אוֹ שֶׁמְּחֻלָּק לִשְׁתֵּי מִזְרָנִים, אִם הָאֶחָד מַרְגִּישׁ שֶׁזֶּה שׁוֹקֵעַ בִּגְלַל כֹּבֶד הַשֵּׁנִי - אָסוּר. אֲבָל אִם בְּסִיס הַמִּטָּה יַצִּיב וְאֵינוֹ שׁוֹקֵעַ - מֻתָּר. וְגַם בְּמִטָּה שׁוֹקַעַת אִם הֵם יוֹשְׁבִים בְּרִחוּק גָּדוֹל כָּךְ שֶׁאֵין הָאֶחָד מַרְגִּישׁ בְּכָבְדּוֹ שֶׁל הַשֵּׁנִי - מֻתָּר.

וּמַתִּירִים (בִּשְׁעַת הַדְּחָק) כְּשֶׁאָדָם יוֹשֵׁב בֵּינֵיהֶם [בִּשְׁעַת הַדְּחָק גָּדוֹל מַתִּירִים גַּם אִם שָׂמִים בֵּינֵיהֶם חֵפֶץ גָּדוֹל, כְּגוֹן תִּיק גָּדוֹל]22.

כג. לְפִיכָךְ, בִּנְסִיעָה בִּמְכוֹנִית אוֹ בְּאוֹטוֹבּוּס וכיו"ב, אִם זֶה מוֹשָׁב אֶחָד - יֵשׁ לִפְעָמִים אִסּוּר מִצַּד הַדִּין [אִם הַמּוֹשָׁב שׁוֹקֵעַ כַּנַּ"ל]. וְאִם הַמּוֹשָׁבִים חֲלוּקִים וְאֵין שׁוֹקְעִים - אֵין אִסּוּר מִצַּד הַדִּין, אִם יְכוֹלִים לְהִזָּהֵר שֶׁלֹּא יִדָּחֲקוּ בְּיַחַד וְגַם לְהִזָּהֵר שֶׁלֹּא יִגְּעוּ הַמַּלְבּוּשִׁים זב"ז.

ע"כ, לְכַתְּחִלָּה יֵשְׁבוּ בִּשְׁנֵי מוֹשָׁבִים [כְּדֵי שֶׁלֹּא יָבוֹאוּ לִנְגִיעָה אוֹ נְגִיעַת הַבְּגָדִים]. אָמְנָם אִם ע"י שֶׁיֵּשְׁבוּ בְּנִפְרָד יֵצֵא שֶׁהַבַּעַל יֵשֵׁב לְיַד אִשָּׁה אַחֶרֶת אוֹ שֶׁהָאִשָּׁה תֵּשֵׁב לְיַד גֶּבֶר אַחֵר - עָדִיף שֶׁיֵּשְׁבוּ בְּנֵי הַזּוּג בְּאוֹתוֹ סַפְסָל [וְיָשִׂימוּ בֵּינֵיהֶם חֵפֶץ כְּדֵי שֶׁלֹּא יָבוֹאוּ לִידֵי נְגִיעָה].

וְכֵן בַּמָּטוֹס, שֶׁהַמּוֹשָׁבִים בּוֹ נִפְרָדִים, אִם אֶפְשָׁר יַשְׁאִירוּ מוֹשָׁב רֵיק בֵּינֵיהֶם. אֲבָל אִם הַמָּטוֹס מָלֵא וְאִם יֵשְׁבוּ בְּנִפְרָד יֵצֵא שֶׁהַבַּעַל יֵשֵׁב לְיַד אִשָּׁה אוֹ שֶׁהָאִשָּׁה תֵּשֵׁב לְיַד אִישׁ, אֲזַי יֵשְׁבוּ סְמוּכִים וְיִזָּהֲרוּ שֶׁלֹּא יִגְּעוּ זֶה בָּזֶה וְשֶׁבִּגְדֵיהֶם לֹא יִגְּעוּ בֵּינֵיהֶם23.
