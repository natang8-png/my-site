כ. אָסוּר לַבַּעַל לֵישֵׁב וְלִשְׁכַּב עַל מִטַּת אִשְׁתּוֹ גַּם שֶׁלֹּא בְּפָנֶיהָ. כֵּן אָסוּר לְהִשְׁתַּמֵּשׁ בַּמַּצָּעִים הַמְיֻחָדִים לָהּ [כְּגוֹן אִם יֵשׁ כַּר אוֹ שְׂמִיכָה הַמְיֻחָדִים לָהּ].

{w}אֲבָל לָאִשָּׁה מֻתָּר לֵישֵׁב עַל מִטָּתוֹ אֲפִלּוּ בְּפָנָיו וְכֵן לֵישֵׁב עַל מַצָּעָיו אֲפִלּוּ בְּפָנָיו, אֲבָל אֲסוּרָה לִשְׁכַּב עַל מִטָּתוֹ בְּפָנָיו. וְאִם הָאִשָּׁה בְּמָקוֹם מְרֻחָק [כְּשָׁעָה נְסִיעָה וָיוֹתֵר] - מֻתָּר לֵישֵׁב עַל מִטָּתָהּ. וְאִם הִיא מֵעֵבֶר לַיָּם אוֹ בְּאֶרֶץ רְחוֹקָה, אוֹ שֶׁנָּסְעָה לְכַמָּה יָמִים מֵהַבַּיִת - מֻתָּר גַּם לִשְׁכַּב עַל מִטָּתָהּ.

"מִטָּה שֶׁלָּהּ" נִקְרֵאת אִם הִיא יְשֵׁנָה עָלֶיהָ בִּקְבִיעוּת אוֹ הִתְחִילָה לִישֹׁן עָלֶיהָ וּבְכַוָּנָתָהּ שֶׁתִּישַׁן בָּהּ בִּקְבִיעוּת. אֲבָל אִם יָשְׁנָה עָלֶיהָ לַיְלָה אֶחָד בְּאַקְרַאי - אֵין אִסּוּר [וְיֵשׁ שֶׁהִתִּירוּ לַבַּעַל לֵישֵׁב עַל מִטַּת אִשְׁתּוֹ הַיּוֹלֶדֶת בְּבֵית־הָרְפוּאָה, אִם הָאִשָּׁה אֵינָהּ בַּמִּטָּה, מִכֵּיוָן שֶׁאי"ז לִזְמַן אָרֹךְ וְאֵינָהּ מְיֹעֶדֶת לְחַיֵּי מִשְׁפָּחָה וְלֹא נִקְרֵאת "מִטָּתָהּ". וְיֵשׁ שֶׁאָסְרוּ גַּם בָּזֶה]20.

כא. מִטָּה שֶׁבַּיּוֹם מִתְקַפֶּלֶת וְהוֹפֶכֶת לְכִסֵּא נוֹחַ וכיו"ב - מֻתָּר לַבַּעַל לֵישֵׁב עָלֶיהָ אַחַר שֶׁקִּפְּלוּהָ לְכִסֵּא.

וְאִם יֵשׁ לָהֶם מִטּוֹת שֶׁמַּכְנִיסִים בַּיּוֹם אֶת שְׁנֵי הַמִּזְרָנִים אֶחָד עַל הַשֵּׁנִי, הֲרֵי אִם הַמִּזְרָן שֶׁל הַבַּעַל לְמַעְלָה - מֻתָּר לֵישֵׁב עָלָיו, וְאִם הַמִּזְרָן שֶׁל הָאִשָּׁה לְמַעְלָה - נֶחְלְקוּ הַפּוֹסְקִים בָּזֶה.

וּמ"מ מֻתָּר לַבַּעַל לִנְגֹּעַ בְּמִטָּתָהּ אוֹ בִּכְלֵי מִטָּתָהּ אוֹ בִּבְגָדֶיהָ, אֲבָל אִם זֶה מֵבִיא לִידֵי הִרְהוּר יִמָּנַע מִכָּךְ [וּלְעִנְיַן הַצָּעַת מִטָּה בִּזְמַן הָאִסּוּר - רְאֵה לְקַמָּן הֲלָכָה כט]21.

  1. רְאֵה סֵפֶר פִּסְקֵי דִּינִים צֶמַח צֶדֶק סי' קצה ס"ה. וּבְסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סְעִיפִים לא-לג, לז.

    וּלְעִנְיַן יְשִׁיבַת הַבַּעַל עַל מִטַּת אִשְׁתּוֹ הַיּוֹלֶדֶת בְּבֵית הָרְפוּאָה - רְאֵה בְּסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פֶּרֶק כח סכ"ד, שֶׁהֵבִיא בָּזֶה מַחְלֹקֶת וּמְסַיֵּם שֶׁיֵּשׁ לְהַחְמִיר. ↑ לטקסט

  2. בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד סל"ב וְאֵילָךְ וּבְסֵפֶר נִטְעֵי גַּבְרִיאֵל הִלְכוֹת נִדָּה פכ"ח סכ"ב הוּבְאָה מַחְלֹקֶת הַאִם מֻתָּר לֵישֵׁב ע"ג סַפָּה כְּפוּלָה שֶׁבַּלַּיְלָה נִפְתַּחַת וּבַיּוֹם נִסְגֶּרֶת וּמַנִּיחִים אֶת מִזְרַן הָאִשָּׁה לְמַעְלָה, אֲבָל אִם מִזְרַן הָאִישׁ הוּא לְמַעְלָה - וַדַּאי מֻתָּר, כְּמ"ש שָׁם בַּהֶעָרָה.

    וְשָׁם בְּסכ"ג מֵבִיא שֶׁסַּפָּה הַמְשַׁמֶּשֶׁת בַּיּוֹם לִישִׁיבַת בְּנֵי הַבַּיִת וּבַלַּיְלָה מְשַׁמֶּשֶׁת לִשְׁנַת הָאִשָּׁה - מֻתָּר לְבַעְלָהּ לֵישֵׁב עָלֶיהָ בַּיּוֹם, אֲבָל אָסוּר לוֹ לִשְׁכַּב עָלֶיהָ בַּיּוֹם. וּבַלַּיְלָה אָסוּר גַּם לֵישֵׁב עַל הַמִּטָּה. ↑ לטקסט
