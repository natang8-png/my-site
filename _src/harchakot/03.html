ב. וְכֵן אֲסוּרִים בְּקֵרוּב פִיזִי כְּמוֹ חִבּוּק וְנִשּׁוּק וְהוּא אִסּוּר לָאו מִן הַתּוֹרָה [וּבִנְגִיעָה שֶׁלֹּא לְשֵׁם הֲנָאָה נֶחְלְקוּ הַפּוֹסְקִים הַאִם אִסּוּרוֹ מִן הַתּוֹרָה אוֹ מִדְּרַבָּנָן]. וּבִכְלַל הָאִסּוּר מִדְּרַבָּנָן גַּם נְגִיעָה ע"י בְּגָדִים אוֹ כְּפָפוֹת, וְאָסוּר גַּם שֶׁבֶּגֶד יִגַּע בְּבֶגֶד וְלָכֵן אָסוּר לָשֶׁבֶת קְרוֹבִים זֶה לָזוֹ אַף שֶׁהַנְּגִיעָה הִיא ע"י הַבְּגָדִים [וְגַם כְּשֶׁאֵין שׁוּם נְגִיעָה בֵּינֵיהֶם, יֵשׁ מַצָּבִים שֶׁאָסְרוּ חֲכָמִים לֵישֵׁב עַל סַפְסָל אֶחָד, כִּדְלְקַמָּן הֲלָכוֹת כב-כג]. אֲבָל בְּמִקְרֶה שֶׁל סַכָּנָה וְכֵן בְּמַצָּבִים שֶׁצְּרִיכָה עֶזְרָה מִיָּדִית דְּחוּפָה, כְּגוֹן בְּמִקְרֶה שֶׁל פְּצִיעָה אוֹ הִתְעַלְּפוּת וכיו"ב – צְרִיכִים לַעֲזֹר לְבֶן אוֹ בַּת הַזּוּג גַּם אִם חַיָּבִים לִגַּע בּוֹ. אֲבָל אִם נִתָּן – יֵשׁ לְהִמָּנַע מִנְּגִיעָה אוֹ לְפָחוֹת לִנְגֹּעַ ע"י מַגֶּבֶת וכיו"ב אוֹ לְפָחוֹת ע"י כְּפָפוֹת [וְהַדִּין בְּחֹלִי שֶׁאֵין הֶכְרֵחַ מִיָּדִי לְעֶזְרָה – רְאֵה לְקַמָּן הֲלָכוֹת לא-לג].

וְרַבָּנָן אָסְרוּ וְגָזְרוּ הַרְבֵּה הַרְחָקוֹת בְּאִשְׁתּוֹ נִדָּה מִפְּנֵי שֶׁרְגִילִים זֶה בָּזוֹ וּבְקַל יָבוֹאוּ לִידֵי חֵטְא וְגַם מִפְּנֵי שֶׁמֻּתָּרִים בְּיִחוּד, כְּלוֹמַר, שֶׁמֻּתָּרִים לִהְיוֹת לְבַדָּם בַּבַּיִת וְיֵשׁ חֲשָׁשׁ שֶׁיִּכָּשְׁלוּ, ע"כ הֶחְמִירוּ בָּהֶם בְּכַמָּה דְּבָרִים כְּדֵי שֶׁלֹּא יָבוֹאוּ לִידֵי מִכְשׁוֹל. וּמָצִינוּ בַּחֲזַ"ל שֶׁעַל חֹסֶר זְהִירוּת בָּזֶה נֶעֱנַשׁ תַּלְמִיד־חָכָם בְּעֹנֶשׁ מִיתָה, ה"י2.

  1. הוּבָא בשוע"ר סי' קפד סק"ו, שֶׁחִבּוּק וְנִשּׁוּק אָסוּר מִן הַתּוֹרָה.

    וִידוּעָה הַמַּחְלֹקֶת בִּנְגִיעָה שֶׁלֹּא לְשֵׁם הֲנָאָה, שֶׁמֵּהב"י בְּסִי' קצה מְדַיְּקִים שֶׁסּוֹבֵר שֶׁכָּל נְגִיעָה הִיא אִסּוּר תּוֹרָה וּנְגִיעָה סְתָמִית אֲסוּרָה רַק מִדְּרַבָּנָן. וּבְטַעַם שֶׁהֶחְמִירוּ בְּיוֹתֵר בְּאִשָּׁה נִדָּה כָּתַב הרא"ש בְּהִלְ' נִדָּה בְּקִצּוּר שֶׁלּוֹ, שֶׁמִּפְּנֵי שֶׁמֻּתָּרִים בְּיִחוּד (רְאֵה סַנְהֶדְרִין לז, א) לָכֵן חָשְׁשׁוּ יוֹתֵר וְאָסְרוּ כָּל הַהַרְחָקוֹת.

    וְהָרשב"א בִּתְשׁוּבוֹתָיו ח"א סי' א'קפח כָּתַב, שֶׁמִּפְּנֵי שֶׁהֵם רְגִילִים זֶה בָּזוֹ לָכֵן חָשְׁשׁוּ יוֹתֵר וְאָסְרוּ כָּל הַהַרְחָקוֹת.

    הַסִּפּוּר עַל הַתַּלְמִיד־חָכָם שֶׁנֶּעֱנַשׁ הוּא מִמַּסֶּכֶת שַׁבָּת יג, א וְהוּבָא בְּבֵית יוֹסֵף שָׁם וּבִשְׁאָר מְחַבְּרִים וְנֶעְתַּק לְעֵיל בַּפְּתִיחָה לְפֶרֶק זֶה. ↩
