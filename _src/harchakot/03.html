א. אִשָּׁה שֶׁנֶּאֶסְרָה עַל בַּעְלָהּ בְּאֶחָד מֵהָאֳפָנִים הַמְבֹאָרִים לְעֵיל בְּפֶרֶק א' – הֲרֵי הִיא אֲסוּרָה בְּבִיאָה וּבִנְגִיעָה [וּמֵחֲמַת זֶה בְּעוֹד דְּבָרִים שֶׁיִּתְבָּאֲרוּ לְהַלָּן]. אַךְ מוּבָן שֶׁיֵּשׁ לְבַעְלָהּ לְהִתְיַחֵס אֵלֶיהָ בְּכָבוֹד וְאַף בְּיֶתֶר רְגִישׁוּת בְּאוֹתוֹ הַזְּמַן בָּאֳפָנִים הַמֻּתָּרִים.

אִסּוּר נִדָּה חָמוּר הוּא עַד מְאֹד וְהָעוֹבֵר עָלָיו בְּמֵזִיד עָנְשׁוֹ כָּרֵת.

וּלְכַמָּה פּוֹסְקִים ה"ז בִּכְלַל "גִּלּוּי עֲרָיוֹת", שֶׁהַדִּין בָּזֶה הוּא "יֵהָרֵג וְאַל יַעֲבֹר". כְּלוֹמַר, שֶׁאִם כּוֹפִים אִישׁ לְשַׁמֵּשׁ עִם אִשְׁתּוֹ נִדָּה וְאִם לֹא יַסְכִּים יַהַרְגוּהוּ – עָלָיו לִמְסֹר נַפְשׁוֹ לְמִיתָה וְלֹא לַעֲבֹר [אָמְנָם אִשָּׁה אֵינָהּ מְחֻיֶּבֶת לִמְסֹר נַפְשָׁהּ, כִּי הִיא אֵינָהּ עוֹשָׂה מַעֲשֶׂה, אַךְ וַדַּאי שֶׁעָלֶיהָ לְהִתְנַגֵּד וְלַעֲשׂוֹת כָּל מַה שֶּׁיְּכוֹלָה שֶׁלֹּא תִּכָּשֵׁל בְּאִסּוּר נִדָּה אוֹ שְׁאָר אִסּוּרֵי גִּלּוּי עֲרָיוֹת (פְּרָט לְמס"נ)]1.

  1. רְאֵה הָרמ"א בְּסוֹף סי' קפג וּבְסוֹף סי' קפה, וּרְאֵה בב"י סי' קצה, שֶׁדָּן מַדּוּעַ מֻתָּר לָרוֹפֵא לְרַפֵּא אִשְׁתּוֹ נִדָּה בַּחֲשַׁשׁ סַכָּנָה אע"פ שֶׁקיי"ל בְּגִלּוּי עֲרָיוֹת יֵהָרֵג וְאַל יַעֲבֹר. וּמִזֶּה מְדַיְּקִים, שֶׁדַּעַת הב"י שֶׁנִּדָּה הִיא בִּכְלַל גִּלּוּי עֲרָיוֹת.

    וּלְהָעִיר מִדִּבְרֵי אדמוה"ז בְּאִגֶּרֶת הַתְּשׁוּבָה פ"ו, שֶׁאֵין לְהַקְשׁוֹת מִכָּךְ שֶׁבִּזְמַן הַגָּלוּת אֶפְשָׁר שֶׁחַיָּבֵי כְּרֵיתוֹת יַאֲרִיכוּ יְמֵיהֶם וּשְׁנוֹתֵיהֶם בַּנְּעִימִים, שֶׁכֵּן הַחַיּוּת שֶׁמְּקַבְּלִים הִיא רַק מִצַּד הַקְּלִפּוֹת. עי"ש. ↩
