טז. אֲסוּרִים לִמְזֹג מַשְׁקֶה אוֹ לְהַגִּישׁ אֶחָד לַשֵּׁנִי מַאֲכָל אוֹ מַשְׁקֶה חוּץ מִמַּיִם.

וְאִם מַגִּישִׁים ע"י שִׁנּוּי, כְּגוֹן בְּיַד שְׂמֹאל בִּמְקוֹם בְּיַד יָמִין [וּשְׂמָאלִי יַגִּישׁ בִּימִין כָּל אָדָם] אוֹ שֶׁמַּנִּיחַ בְּרִחוּק קְצָת שֶׁלֹּא בַּמָּקוֹם שֶׁרְגִילִים לְהַנִּיחַ - מֻתָּר [לְעִנְיַן מְזִיגָה בְּרִחוּק - רְאֵה בַּהֶעָרָה]. וְכֵן מֻתָּר לְהַגִּישׁ בִּמְקוֹמוֹ בְּשָׁעָה שֶׁבֶּן הַזּוּג אֵינוֹ נִמְצָא שָׁם16.

יז. מֻתָּר לְהַגִּישׁ צַלַּחַת מֶרְכָּזִית לְכַמָּה סוֹעֲדִים וּבֵינֵיהֶם גַּם בֶּן אוֹ בַּת הַזּוּג, אֲבָל אֵין לְהַנִּיחָהּ בְּיִחוּד לִפְנֵי בֶּן הַזּוּג כְּדֵי שֶׁיִּבְרֹר לוֹ מָנָה [אע"פ שֶׁאח"כ יַעֲבִירוּ אֶת הַצַּלַּחַת כְּדֵי שֶׁיִּקְחוּ שְׁאָר הַמְסֻבִּים מִצַּלַּחַת זוֹ]17.

יח. בִּזְמַן טֻמְאָתָהּ אָסוּר לַבַּעַל לִשְׁלֹחַ [כְּגוֹן ע"י יֶלֶד] לְאִשְׁתּוֹ כּוֹס יַיִן אוֹ מַשְׁקֶה אַחֵר [פְּרָט לְמַיִם כִּדְלְהַלָּן]. וְלָכֵן אֶת כּוֹס הַקִּדּוּשׁ יַשְׁאִיר עַל הַשֻּׁלְחָן וְלֹא יִשְׁלַח לָהּ. אֲבָל מֻתָּר לִשְׁלֹחַ לָהּ מַאֲכָל אוֹ מַיִם [אִם הִגִּישׁוּ אוֹ שָׁלְחוּ בְּדִיעֲבַד - רְאֵה בַּהֶעָרָה].

וְאִם שׁוֹלֵחַ יַיִן לַמְסֻבִּים וּבְהֶמְשֵׁךְ יַגִּיעַ הַכּוֹס לְאִשְׁתּוֹ - מֻתָּר.

לָאִשָּׁה מֻתָּר לִשְׁלֹחַ לְבַעְלָהּ ע"י שָׁלִיחַ כָּל דָּבָר וַאֲפִלּוּ יַיִן וּמַשְׁקֶה חָשׁוּב18.

    וּלְעִנְיַן מְזִיגָה מֵרָחוֹק - הַדָּבָר מֻתָּר לְרֹב הַפּוֹסְקִים. וְיֵשׁ שֶׁאָסְרוּ מְזִיגָה שֶׁהִיא עִרְבּוּב דְּבָרִים, כְּגוֹן הַכְנָסַת קָפֶה עִם סֻכָּר וְחָלָב בִּפְנֵי הַבַּעַל גַּם אִם מוֹזְגִים מֵרָחוֹק. אֲבָל אִם אֵין זוֹ מְזִיגָה מֻשְׁלֶמֶת, כְּגוֹן בְּאֹפֶן שֶׁלְּבַסּוֹף יִצְטָרֵךְ לְהוֹסִיף אֶת הֶחָלָב אוֹ הַסֻּכָּר - מֻתָּר לְכָל הַדֵּעוֹת. וְהוּבְאָה חֻמְרָא זוֹ בְּסֵפֶר טָהֳרָה כַּהֲלָכָה פי"ד הֶעָרָה 198, וּמוּבָן שָׁם שֶׁזּוֹ חֻמְרָא וְאֵין בָּזֶה אִסּוּר מִן הַדִּין. וְאִם מוֹזְגִים כַּמָּה כּוֹסוֹת וְרַק אַחַת מֵהֶן מְיֹעֶדֶת לְבֶן אוֹ לְבַת הַזּוּג, וְלֹא בָּרוּר אֵיזוֹ מֵהֶן - יֵשׁ מַתִּירִים לִמְזֹג וְיֵשׁ שֶׁאוֹסְרִים לִמְזֹג אִם אֵינָם מוֹזְגִים בְּשִׁנּוּי, כְּגוֹן בְּיַד שְׂמֹאל אוֹ בְּרִחוּק מָקוֹם, כַּנַּ"ל. ↑ לטקסט
