לא. אִם הַבַּעַל הוּא חוֹלֶה וְתָשׁוּשׁ, כָּךְ שֶׁאֵין חֲשָׁשׁ שֶׁיָּבוֹא לְתשהמ"ט בְּאִסּוּר, וְאֵין מִי שֶׁיַּעֲזֹר לוֹ אֶלָּא אִשְׁתּוֹ – מֻתֶּרֶת לְשַׁמְּשׁוֹ, כְּגוֹן לַהֲקִימוֹ אוֹ לְהַשְׁכִּיבוֹ וכיו"ב. וְתִשְׁתַּדֵּל לֹא לִנְגֹּעַ בְּגוּפוֹ אֶלָּא בְּאֶמְצָעוּת מַגֶּבֶת וכיו"ב אוֹ לְפָחוֹת ע"י בֶּגֶד אוֹ כְּפָפוֹת וּלְלֹא מַגָּע בִּבְשָׂרוֹ.

וְכֵן מֻתֶּרֶת לִמְזֹג וּלְהַגִּישׁ לוֹ כּוֹס אוֹ מַאֲכָל, אֲבָל תִּשְׁתַּדֵּל לַעֲשׂוֹת זֹאת בְּאֹפֶן הַמֻּתָּר, כְּגוֹן בְּיַד שְׂמֹאל אוֹ בְּרִחוּק קְצָת וְכַנַּ"ל, אוֹ שֶׁהוּא לֹא יַבִּיט בִּזְמַן הַהַגָּשָׁה, אֲבָל אִם אֵין אֶפְשָׁרוּת אַחֶרֶת – מֻתֶּרֶת בַּכֹּל.

וְתִשְׁתַּדֵּל לְהִמָּנַע מֵרְחִיצַת פָּנָיו יָדָיו וְרַגְלָיו וְהַצָּעַת הַמִּטָּה בְּפָנָיו. וְאִם יֵשׁ לוֹ צֹרֶךְ בָּזֶה – תִּשְׁתַּדֵּל עכ"פ לִרְחֹץ ע"י כְּפָפוֹת וכיו"ב וּלְהַצִּיעַ הַמִּטָּה שֶׁלֹּא בְּפָנָיו אוֹ עכ"פ שֶׁלֹּא יַבִּיט, כַּנַּ"ל.

וְכָל זֶה, כָּאָמוּר, בְּאָדָם חוֹלֶה וְתָשׁוּשׁ, אֲבָל אָדָם שֶׁמְּתַפְקֵד כָּרָגִיל וְיֵשׁ לוֹ רַק כְּאֵב מְקוֹמִי וְכֵן אִם יֵשׁ לוֹ שֶׁבֶר וכד' אֲבָל בִּכְלָלוּת מַרְגִּישׁ טוֹב – דִּינוֹ כְּבָרִיא. וְכֵן אִם יֵשׁ לוֹ מֵחוּשׁ בִּכְלָלוּת הַגּוּף, כְּגוֹן כְּאֵב רֹאשׁ קַל שֶׁנִּתָּן לְסָבְלוֹ וּלְתַפְקֵד כָּרָגִיל אוֹ שֶׁנָּטַל רְפוּאוֹת וְעי"ז נִרְגַּע הַכְּאֵב הֶחָזָק וְיָכוֹל לְתַפְקֵד כָּרָגִיל אוֹ שֶׁיֵּשׁ לוֹ מְעַט חֹם וּמְתַפְקֵד כָּרָגִיל – דִּינוֹ כְּבָרִיא לְעִנְיָן זֶה (וְאִם הַבַּעַל מֻגְבָּל בִּתְנוּעוֹתָיו, כְּגוֹן עֵקֶב פְּצִיעָה וְזָקוּק מְאֹד לְעֶזְרַת הָאִשָּׁה אַךְ אֵינוֹ מֻגְדָּר כְּחוֹלֶה, יֵשׁ לִשְׁאֹל רַב).

בְּמָקוֹם שֶׁמֻּתָּר לָאִשָּׁה לַעֲזֹר לְבַעַל חוֹלֶה – אֵין צְרִיכִים לִשְׂכֹּר בְּכֶסֶף גֶּבֶר שֶׁיְּטַפֵּל בַּבַּעַל, אֲבָל אִם נִתָּן לְבַקֵּשׁ וּלְקַבֵּל עֶזְרָה מִגֶּבֶר לְלֹא תַּשְׁלוּם – צְרִיכָה לְבַקֵּשׁ זֶה גַּם אִם זֶה כָּרוּךְ בְּאִי נְעִימוּת31.

לב. אִם הָאִשָּׁה חוֹלָה – אָסוּר לַבַּעַל לְשַׁמְּשָׁהּ אֶלָּא אִם כֵּן זְקוּקָה מְאֹד לְעֶזְרָתוֹ. וְכַמּוּבָן בְּמִקְרֶה שֶׁנִּפְצְעָה אוֹ הִתְעַלְּפָה וּזְקוּקָה מִיָּדִית לְעֶזְרָתוֹ – וַדַּאי שֶׁצָּרִיךְ לְעוֹזְרָהּ.

אֲבָל כְּשֶׁיָּדוּעַ שֶׁתִּזְדַּקֵּק לְסִיּוּעַ בִּזְמַן מְסֻיָּם, עַל הַבַּעַל לִדְאֹג שֶׁאִשָּׁה תְּטַפֵּל בָּהּ גַּם אִם זֶה כָּרוּךְ בְּתַשְׁלוּם כַּסְפִּי וְאִם קָשֶׁה עָלָיו הַהוֹצָאָה הַכַּסְפִּית אוֹ שֶׁיֵּשׁ בָּזֶה קֹשִׁי אַחֵר – יִשְׁאַל רַב32.

לג. אִם הַבַּעַל רוֹפֵא אוֹ אָח רְפוּאִי – לֹא יְטַפֵּל בְּאִשְׁתּוֹ בִּימֵי נִדָּתָהּ, אֲבָל אִם יֵשׁ חֲשַׁשׁ סַכָּנָה וְאֵין שָׁם רוֹפֵא אַחֵר אוֹ שֶׁבַּעְלָהּ בָּקִי יוֹתֵר בָּרְפוּאָה – מֻתָּר [וּמוּבָן שֶׁמֻּתָּר לַבַּעַל לְהַגִּישׁ כָּל סִיּוּעַ דָּחוּף, כְּגוֹן אִם נִתְעַלְּפָה אוֹ נִפְצְעָה, כַּנַּ"ל הֲלָכָה לב]33.
