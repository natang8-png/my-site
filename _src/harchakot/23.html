לד. נָהֲגוּ הַנָּשִׁים שֶׁלֹּא לֵילֵךְ לְבֵית הַקְּבָרוֹת אוֹ לְהַלְוָיָה בִּימֵי נִדָּתָן וּלְכַתְּחִלָּה נִמְנָעוֹת עַד אַחַר הַטָּהֳרָה בַּמִּקְוֶה וּלְפָחוֹת מַקְפִּידוֹת לְהִמָּנַע בִּזְמַן הַדִּמּוּם, וּמ"מ, בְּצֹרֶךְ גָּדוֹל מַתִּירִים. אֲבָל יֵשׁ לָהּ לְהִשְׁתַּדֵּל לְהַרְחִיק ד' אַמּוֹת מֵהַמִּטָּה אוֹ מֵהַקֶּבֶר. וְכֵן יֵשׁ לָהּ לְהִשְׁתַּדֵּל שֶׁלֹּא לְהִכָּנֵס לְתוֹךְ ד' אַמּוֹת שֶׁל קֶבֶר צַדִּיק וְלֹא לְתוֹךְ אֹהֶל צַדִּיק גַּם בְּרִחוּק ד' אַמּוֹת מִקֶּבֶר צַדִּיק בִּזְמַן הַדִּמּוּם עכ"פ, אַךְ לַכֹּתֶל הַמַּעֲרָבִי נָהֲגוּ לְהָקֵל [כִּי לְרֹב הַפּוֹסְקִים הוּא כֹּתֶל הַר הַבַּיִת וְלֹא כֹּתֶל הָעֲזָרָה].

וְכֵן נָהֲגוּ הַנָּשִׁים בִּזְמַן הַהֵרָיוֹן שֶׁלֹּא לֵילֵךְ לְהַלְוָיָה אוֹ לְבֵית הַקְּבָרוֹת וּלְצֹרֶךְ גָּדוֹל מַתִּירִים רַק עָלֶיהָ לְהִשְׁתַּדֵּל לְהַרְחִיק ד' אַמּוֹת מֵהַקֶּבֶר אוֹ מֵהַמֵּת, אֲבָל אֵין מְנִיעָה בִּזְמַן הַהֵרָיוֹן לְבַקֵּר בְּקִבְרֵי צַדִּיקִים שֶׁאֵינָם בְּבֵית עָלְמִין.

אֵשֶׁת כֹּהֵן מְעֻבֶּרֶת - מִן הַדִּין מֻתֶּרֶת לְהִטַּמֵּא, אַךְ יֵשׁ שֶׁנָּהֲגוּ לְהַחְמִיר בָּזֶה בִּפְרָט כְּשֶׁיָּדוּעַ שֶׁהָעֻבָּר זָכָר. וְכֵן מַחְמִירוֹת שֶׁלֹּא לֵילֵךְ לְקִבְרֵי צַדִּיקִים בִּזְמַן הַהֵרָיוֹן [וְכֵן יֵשׁ לְאֵשֶׁת כֹּהֵן לֵילֵד בְּבֵית־רְפוּאָה שֶׁמֻּחְזָק שֶׁאֵין שָׁם טֻמְאַת מֵת (אִם הוּא בְּרָמָה רְפוּאִית טוֹבָה), כְּדֵי שֶׁלֹּא לְטַמֵּא אֶת הַנּוֹלָד כֹּהֵן בְּטֻמְאַת מֵת. וְכַמּוּבָן בְּכָל מַצָּב דָּחוּף יֵשׁ לִפְנוֹת לְבֵית־הָרְפוּאָה הַקָּרוֹב, וְאִם יֵשׁ צֹרֶךְ, מֻתָּר לְבַעְלָהּ הַכֹּהֵן לֵילֵךְ עִמָּהּ לְשָׁם, אַךְ אִם רוֹצֶה לִשְׁהוֹת שָׁם אוֹ לְבַקְּרָהּ שָׁם - יִשְׁאַל רַב]34.
